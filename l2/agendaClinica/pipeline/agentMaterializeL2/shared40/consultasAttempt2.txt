private resetVisit(): void { this.publish('pageStatus', 'idle'); this.publish('scenary', ''); this.publish('agendarConsultaDraft', EMPTY_DRAFT()); this.publish('filtrarAgendaStatus', 'idle'); this.publish('filtrarAgendaError', null); this.publish('carregarMaisAgendaStatus', 'idle'); this.publish('carregarMaisAgendaError', null); this.publish('consultarConsultaSelecionadaStatus', 'idle'); this.publish('consultarConsultaSelecionadaError', null); this.publish('localizarPacientesParaAgendamentoStatus', 'idle'); this.publish('localizarPacientesParaAgendamentoError', null); this.publish('carregarMaisPacientesParaAgendamentoStatus', 'idle'); this.publish('carregarMaisPacientesParaAgendamentoError', null); this.publish('localizarProfissionaisParaAgendamentoStatus', 'idle'); this.publish('localizarProfissionaisParaAgendamentoError', null); this.publish('carregarMaisProfissionaisParaAgendamentoStatus', 'idle'); this.publish('carregarMaisProfissionaisParaAgendamentoError', null); this.publish('agendarConsultaStatus', 'idle'); this.publish('agendarConsultaError', null); this.publish('confirmarConsultaStatus', 'idle'); this.publish('confirmarConsultaError', null); this.publish('registrarFaltaStatus', 'idle'); this.publish('registrarFaltaError', null); }
private applyEntryParams(): void { const params = new URLSearchParams(window.location.search); const names = ['dataAgenda', 'status', 'page', 'id', 'termo', 'pacienteId', 'consultaId', 'profissionalId']; for (const name of names) { let raw = params.get(name); if (raw === null) { try { raw = localStorage.getItem(`agendaClinica.consultas.${name}`); } catch { raw = null; } } if (name === 'page') { const numberValue = raw === null ? null : Number(raw); if (numberValue !== null && Number.isFinite(numberValue)) this.publish('page', numberValue); } else if (name === 'consultaId') { this.publish('consultaId', raw); this.publish('selectedConsulta', raw); } else if (name === 'dataAgenda') this.publish('dataAgenda', raw); else if (name === 'status') this.publish('status', raw); else if (name === 'id') this.publish('id', raw); else if (name === 'termo') this.publish('termo', raw); else if (name === 'pacienteId') this.publish('pacienteId', raw); else if (name === 'profissionalId') { this.publish('profissionalId', raw); const draft = this.agendarConsultaDraft; this.publish('agendarConsultaDraft', { ...draft, profissionalId: raw }); } } }
public handleIcaStateChange(key: string, value: any): void { if (!STATE_MEMBER_BY_KEY[key]) { super.handleIcaStateChange(key, value); return; } if (value === undefined) return; this.assignState(key, value); this.requestUpdate(); }
