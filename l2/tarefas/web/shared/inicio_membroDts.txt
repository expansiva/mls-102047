import { StateLitElement } from '/_102029_/l2/stateLitElement.js';
import type { ResumoMinhasTarefas, AvisoDoDia, EsperaPorMim } from '/_102047_/l2/tarefas/web/contracts/inicio_membro.defs.js';
export type { Inicio_membroContracts, ResumoMinhasTarefas, AvisoDoDia, EsperaPorMim, TarefaResumoDetalhes, ListaRef, TarefaResumo, AvisoTarefa, AvisoDetalhes, MembroIdentificacao, MembroDetalhes, MembroRef, TarefaPendenciaDetalhes, TarefaPendencia } from '/_102047_/l2/tarefas/web/contracts/inicio_membro.defs.js';
export type ErrorState = {
    code: string;
    message: string;
    details?: unknown;
} | null;
export type PageStatus = 'idle' | 'loading' | 'empty' | 'success' | 'error';
export type RequestStatus = 'idle' | 'loading' | 'success' | 'error';
/** Draft of the devolverComMotivo command input; every leaf is null until filled. */
export type DevolverComMotivoDraft = {
    id: string | null;
    version: number | null;
    details: {
        motivoDevolucao: string | null;
    };
};
export declare class TarefasInicio_membroShared extends StateLitElement {
    /** state resumo — workload of the person: totals of open, due-today and overdue tasks and the most urgent open tasks; source carregarInicio.resumo; organism resumoMinhasTarefas */
    resumo: ResumoMinhasTarefas | null;
    /** state avisoDoDia — daily due-date notice of the person for today, with delivery status and included tasks (null when not generated yet); source carregarInicio.avisoDoDia; organism avisoDoDia */
    avisoDoDia: AvisoDoDia | null;
    /** state esperaPorMim — tasks that depend on the person grouped by expected action: start, send to review, review, adjust after return; source carregarInicio.esperaPorMim; organism esperaPorMim */
    esperaPorMim: EsperaPorMim | null;
    /** state devolverComMotivoDraft — draft of the return-with-reason command (reason typed by the reviewer; id and version are filled when the command is submitted); source devolverComMotivo.input; organism esperaPorMim */
    devolverComMotivoDraft: DevolverComMotivoDraft;
    /** state pageStatus — overall status of the page: idle, loading, empty (every list empty), success or error; written only by the initial load */
    pageStatus: PageStatus;
    /** state carregarInicioStatus — status of the carregarInicio request */
    carregarInicioStatus: RequestStatus;
    /** state carregarInicioError — error of the carregarInicio request (also client errors of navigations), null when none */
    carregarInicioError: ErrorState;
    /** state iniciarTarefaStatus — status of the iniciarTarefa command */
    iniciarTarefaStatus: RequestStatus;
    /** state iniciarTarefaError — error of the iniciarTarefa command, null when none */
    iniciarTarefaError: ErrorState;
    /** state enviarParaRevisaoStatus — status of the enviarParaRevisao command */
    enviarParaRevisaoStatus: RequestStatus;
    /** state enviarParaRevisaoError — error of the enviarParaRevisao command, null when none */
    enviarParaRevisaoError: ErrorState;
    /** state aceitarEntregaStatus — status of the aceitarEntrega command */
    aceitarEntregaStatus: RequestStatus;
    /** state aceitarEntregaError — error of the aceitarEntrega command, null when none */
    aceitarEntregaError: ErrorState;
    /** state devolverComMotivoStatus — status of the devolverComMotivo command */
    devolverComMotivoStatus: RequestStatus;
    /** state devolverComMotivoError — error of the devolverComMotivo command, null when none */
    devolverComMotivoError: ErrorState;
    /** state scenary — the visible scene of the page; '' until the page shows its first scene */
    scenary: string;
    private subscribed;
    private publish;
    private assignState;
    private setRequestStatus;
    private setRequestError;
    private failRequest;
    private unexpectedError;
    private resetCommandResults;
    private resetVisit;
    private hydrateData;
    connectedCallback(): void;
    disconnectedCallback(): void;
    handleIcaStateChange(key: string, value: any): void;
    /** scene — publishes the visible scene of the page (any string; '' means no scene yet); clears the results of the commands */
    setScenario(value: string): void;
    /** draft — replaces the draft of the return-with-reason command (the reviewer types details.motivoDevolucao); clears the results of the commands */
    setDevolverComMotivoDraft(value: DevolverComMotivoDraft): void;
    /** request carregarInicio — loads the whole Início of the member: workload summary, today's daily notice and the pending tasks waiting for the person; redraws resumo, avisoDoDia and esperaPorMim (replace); no arguments, the person is the session's. Also the method to reload the page. */
    carregarInicio(): Promise<void>;
    private applyCommandOutput;
    private runTransition;
    private getRequestStatus;
    /** request iniciarTarefa — starts a task assigned to the person (aFazer → emAndamento); redraws resumo and esperaPorMim (replace with the recalculated values); arguments: id = task to start, version = version read from the row (optimistic concurrency) */
    iniciarTarefa(id: string, version: number): Promise<void>;
    /** request enviarParaRevisao — sends an in-progress task of the person (also a returned one) to the creator's review; redraws resumo and esperaPorMim (replace with the recalculated values); arguments: id = task to send, version = version read from the row */
    enviarParaRevisao(id: string, version: number): Promise<void>;
    /** request aceitarEntrega — accepts the delivery of a task created by the person, completing it; redraws resumo and esperaPorMim (replace; the task leaves the to-review list); arguments: id = task in review, version = version read from the row */
    aceitarEntrega(id: string, version: number): Promise<void>;
    /** request devolverComMotivo — returns a task in review created by the person for adjustment, with the reason typed in devolverComMotivoDraft.details.motivoDevolucao; redraws resumo and esperaPorMim (replace; the task leaves the to-review list) and clears the draft; arguments: id = task in review, version = version read from the row */
    devolverComMotivo(id: string, version: number): Promise<void>;
    /** navigate — opens the person's task list (minhas_tarefas); carries nothing */
    abrirMinhasTarefas(): void;
    /** navigate — opens the daily notices page (meus_avisos), carrying today's notice id as avisoVencimentoId when the notice exists */
    abrirMeusAvisos(): void;
    /** navigate — opens minhas_tarefas with the picked task selected (tarefaId); argument: tarefaId = id of the task row the person opens; null is refused with client.preconditionMissing in carregarInicioError */
    abrirTarefaParaAjustar(tarefaId: string | null): void;
}
