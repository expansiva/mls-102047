    switch (id) {

    this.publish('iniciarTarefaStatus', 'idle');
    this.publish('iniciarTarefaError', null);
    this.publish('enviarParaRevisaoStatus', 'idle');
    this.publish('enviarParaRevisaoError', null);
    this.publish('aceitarEntregaStatus', 'idle');
    this.publish('aceitarEntregaError', null);
    this.publish('devolverComMotivoStatus', 'idle');
    this.publish('devolverComMotivoError', null);
  }

    this.publish('pageStatus', 'idle');
    this.publish('carregarInicioStatus', 'idle');
    this.publish('carregarInicioError', null);
    this.resetCommandResults();
    this.publish('devolverComMotivoDraft', createEmptyDevolverComMotivoDraft());
    this.publish('scenary', '');
  }

  private hydrateData(): void {
    const resumo = getState('ui.tarefas.inicio_membro.resumo');
    if (resumo !== undefined) this.assignState('ui.tarefas.inicio_membro.resumo', resumo);
    const aviso = getState('ui.tarefas.inicio_membro.avisoDoDia');
    if (aviso !== undefined) this.assignState('ui.tarefas.inicio_membro.avisoDoDia', aviso);
    const espera = getState('ui.tarefas.inicio_membro.esperaPorMim');
    if (espera !== undefined) this.assignState('ui.tarefas.inicio_membro.esperaPorMim', espera);
  }

    this.resetVisit();
    if (!this.subscribed) {
      subscribe([...STATE_KEYS], this);

    }
    void this.carregarInicio();
  }

  /** request carregarInicio — loads the whole Início of the member: workload summary, today's daily notice and the pending tasks waiting for the person; redraws resumo, avisoDoDia and esperaPorMim (replace); no arguments, the person is the session's. Also the method to reload the page. */
  public async carregarInicio(): Promise<void> {
    if (this.carregarInicioStatus === 'loading') return;
    this.publish('carregarInicioStatus', 'loading');
    this.publish('carregarInicioError', null);
    this.publish('pageStatus', 'loading');
    const route = 'tarefas.inicio_membro.carregarInicio';
    const input: Input<typeof route> = {};
    try {
      const response = await execBff<Output<typeof route>>(route, input, { mode: 'silent' });
      if (!response.ok || !response.data) {
        this.publish('carregarInicioError', response.error);
        this.publish('carregarInicioStatus', 'error');
        this.publish('pageStatus', 'error');
        return;
      }
      const data = response.data;
      this.publish('resumo', data.resumo);
      this.publish('avisoDoDia', data.avisoDoDia ?? null);
      this.publish('esperaPorMim', data.esperaPorMim);
      const espera = data.esperaPorMim;
      const isEmpty =
        data.resumo.tarefas.length === 0 &&
        espera.paraIniciar.length === 0 &&
        espera.paraEnviarRevisao.length === 0 &&
        espera.paraRevisar.length === 0 &&
        espera.devolvidas.length === 0;
      this.publish('carregarInicioStatus', 'success');
      this.publish('pageStatus', isEmpty ? 'empty' : 'success');
    } catch (error) {
      this.publish('carregarInicioError', this.unexpectedError(error));
      this.publish('carregarInicioStatus', 'error');
      this.publish('pageStatus', 'error');
    }
  }

  private applyCommandOutput(output: TransitionOutput): void {
    this.publish('resumo', output.resumo);
    this.publish('esperaPorMim', output.esperaPorMim);
  }

  private async runTransition(requestId: TransitionRequestId, route: TransitionRoute, id: string, version: number): Promise<void> {
    if (this.getRequestStatus(requestId) === 'loading') return;
    this.setRequestStatus(requestId, 'loading');
    this.setRequestError(requestId, null);
    if (!id) {
      this.failRequest(requestId, { code: 'client.requiredMissing', message: '', details: { stateKeys: ['id'] } });
      return;
    }
    const input: Input<TransitionRoute> = { id, version };
    try {
      const response = await runBlockingUiAction(signal => execBff<TransitionOutput>(route, input, { mode: 'blocking', signal }));
      if (response === undefined) {
        this.failRequest(requestId, { code: 'client.unexpected', message: '', details: { name: 'aborted' } });
        return;
      }
      if (!response.ok || !response.data) {
        this.failRequest(requestId, response.error);
        return;
      }
      this.applyCommandOutput(response.data);
      this.setRequestStatus(requestId, 'success');
    } catch (error) {
      this.failRequest(requestId, this.unexpectedError(error));
    }
  }

  private getRequestStatus(requestId: RequestId): RequestStatus {
    switch (requestId) {
      case 'carregarInicio':
        return this.carregarInicioStatus;
      case 'iniciarTarefa':
        return this.iniciarTarefaStatus;
      case 'enviarParaRevisao':
        return this.enviarParaRevisaoStatus;
      case 'aceitarEntrega':
        return this.aceitarEntregaStatus;
      case 'devolverComMotivo':
        return this.devolverComMotivoStatus;
    }
  }

  /** request iniciarTarefa — starts a task assigned to the person (aFazer → emAndamento); redraws resumo and esperaPorMim (replace with the recalculated values); arguments: id = task to start, version = version read from the row (optimistic concurrency) */
  public async iniciarTarefa(id: string, version: number): Promise<void> {
    await this.runTransition('iniciarTarefa', 'tarefas.inicio_membro.iniciarTarefa', id, version);
  }

  /** request enviarParaRevisao — sends an in-progress task of the person (also a returned one) to the creator's review; redraws resumo and esperaPorMim (replace with the recalculated values); arguments: id = task to send, version = version read from the row */
  public async enviarParaRevisao(id: string, version: number): Promise<void> {
    await this.runTransition('enviarParaRevisao', 'tarefas.inicio_membro.enviarParaRevisao', id, version);
  }

  /** request aceitarEntrega — accepts the delivery of a task created by the person, completing it; redraws resumo and esperaPorMim (replace; the task leaves the to-review list); arguments: id = task in review, version = version read from the row */
  public async aceitarEntrega(id: string, version: number): Promise<void> {
    await this.runTransition('aceitarEntrega', 'tarefas.inicio_membro.aceitarEntrega', id, version);
  }

  /** request devolverComMotivo — returns a task in review created by the person for adjustment, with the reason typed in devolverComMotivoDraft.details.motivoDevolucao; redraws resumo and esperaPorMim (replace; the task leaves the to-review list) and clears the draft; arguments: id = task in review, version = version read from the row */
  public async devolverComMotivo(id: string, version: number): Promise<void> {
    if (this.devolverComMotivoStatus === 'loading') return;
    this.publish('devolverComMotivoStatus', 'loading');
    this.publish('devolverComMotivoError', null);
    const motivoRaw = this.devolverComMotivoDraft.details.motivoDevolucao;
    const motivo = motivoRaw === null ? '' : motivoRaw.trim();
    const missing: string[] = [];
    if (!id) missing.push('ui.tarefas.inicio_membro.devolverComMotivoDraft.id');
    if (motivo === '') missing.push('ui.tarefas.inicio_membro.devolverComMotivoDraft.details.motivoDevolucao');
    if (missing.length > 0) {
      this.failRequest('devolverComMotivo', { code: 'client.requiredMissing', message: '', details: { stateKeys: missing } });
      return;
    }
    const route = 'tarefas.inicio_membro.devolverComMotivo';
    const input: Input<typeof route> = { id, version, details: { motivoDevolucao: motivo } };
    try {
      const response = await runBlockingUiAction(signal => execBff<Output<typeof route>>(route, input, { mode: 'blocking', signal }));
      if (response === undefined) {
        this.failRequest('devolverComMotivo', { code: 'client.unexpected', message: '', details: { name: 'aborted' } });
        return;
      }
      if (!response.ok || !response.data) {
        this.failRequest('devolverComMotivo', response.error);
        return;
      }
      this.applyCommandOutput(response.data);
      this.publish('devolverComMotivoDraft', createEmptyDevolverComMotivoDraft());
      this.publish('devolverComMotivoStatus', 'success');
    } catch (error) {
      this.failRequest('devolverComMotivo', this.unexpectedError(error));
    }
  }

  /** navigate — opens the person's task list (minhas_tarefas); carries nothing */
  public abrirMinhasTarefas(): void {
    auraNavigate('/tarefas/minhas_tarefas', { basePath: '/tarefas' });
  }

  /** navigate — opens the daily notices page (meus_avisos), carrying today's notice id as avisoVencimentoId when the notice exists */
  public abrirMeusAvisos(): void {
    const avisoId = this.avisoDoDia ? this.avisoDoDia.id : '';
    if (avisoId) {
      auraNavigate(`/tarefas/meus_avisos?avisoVencimentoId=${encodeURIComponent(avisoId)}`, { basePath: '/tarefas' });
      return;
    }
    auraNavigate('/tarefas/meus_avisos', { basePath: '/tarefas' });
  }

  /** navigate — opens minhas_tarefas with the picked task selected (tarefaId); argument: tarefaId = id of the task row the person opens; null is refused with client.preconditionMissing in carregarInicioError */
  public abrirTarefaParaAjustar(tarefaId: string | null): void {
    if (tarefaId === null || tarefaId === '') {
      this.publish('carregarInicioError', { code: 'client.preconditionMissing', message: '', details: { params: ['tarefaId'] } });
      return;
    }
    auraNavigate(`/tarefas/minhas_tarefas?tarefaId=${encodeURIComponent(tarefaId)}`, { basePath: '/tarefas' });
  }
}
