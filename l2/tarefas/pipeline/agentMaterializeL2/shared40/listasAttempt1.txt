export type ErrorState = { code: string; message: string; details?: unknown } | null;
export type RequestStatus = 'idle' | 'loading' | 'success' | 'error';
export type PageStatus = 'idle' | 'loading' | 'empty' | 'success' | 'error';
export type ListaTipo = 'project' | 'area';

/** Details of a list draft; every leaf is null until the person fills it. */
export type ListaDetalhesDraft = {
  tipo: ListaTipo | null;
  descricao: string | null;
};

/** Draft of the criarLista command (a new list). */
export type CriarListaDraft = {
  nome: string | null;
  details: ListaDetalhesDraft;
  ativa: boolean | null;
};

/** Draft of the salvarLista command (editing an existing list, prefilled from the selected row). */
export type SalvarListaDraft = {
  id: string | null;
  version: number | null;
  nome: string | null;
  details: ListaDetalhesDraft;
  ativa: boolean | null;
};

function emptyCriarListaDraft(): CriarListaDraft {
  return { nome: null, details: { tipo: null, descricao: null }, ativa: null };
}

function emptySalvarListaDraft(): SalvarListaDraft {
  return { id: null, version: null, nome: null, details: { tipo: null, descricao: null }, ativa: null };
}

function draftFromLista(lista: ListaCadastrada): SalvarListaDraft {
  return {
    id: lista.id,
    version: lista.version,
    nome: lista.nome,
    details: {
      tipo: lista.details.tipo,
      descricao: lista.details.descricao !== undefined ? lista.details.descricao : null,
    },
    ativa: lista.ativa,
  };
}

function buildDetails(details: ListaDetalhesDraft, tipo: ListaTipo): { tipo: ListaTipo; descricao?: string } {
  const descricao = details.descricao;
  return {
    tipo,
    ...(descricao !== null && descricao !== '' ? { descricao } : {}),
  };
}

export class TarefasListasShared extends StateLitElement {
  /** state listas — the lists of tasks (projects or areas) with editable data, version and open-task count; source listarListas.listas; organism formularioLista, listasCadastradas */
  @property({ attribute: false }) listas: ListaCadastrada[] = [];

  /** state selectedLista — the id of the Lista selected in listasCadastradas (the row is read through the formularioLista getter); source entry.params.listaId; organism formularioLista, listasCadastradas; persisted */
  @property({ attribute: false }) selectedLista: string | null = null;

  /** state busca — the name fragment last used to search the lists ('' lists all); filter held by the page for buscarListas; source buscarListas.input */
  @property({ attribute: false }) busca = '';

  /** state criarListaDraft — draft of the form that creates a new list (nome, details.tipo, details.descricao, ativa); source criarLista.input; organism formularioLista */
  @property({ attribute: false }) criarListaDraft: CriarListaDraft = emptyCriarListaDraft();

  /** state salvarListaDraft — draft of the form that edits the selected list, prefilled from it (id, version, nome, details, ativa); source salvarLista.input; organism formularioLista */
  @property({ attribute: false }) salvarListaDraft: SalvarListaDraft = emptySalvarListaDraft();

  /** state pageStatus — overall status of the page load: idle, loading, empty, success or error; written only by the initial load */
  @property({ attribute: false }) pageStatus: PageStatus = 'idle';

  /** state listarListasStatus — status of the request listarListas */
  @property({ attribute: false }) listarListasStatus: RequestStatus = 'idle';

  /** state listarListasError — error of the request listarListas (also client.selectionInvalid when the selected id matches no row); null when none */
  @property({ attribute: false }) listarListasError: ErrorState = null;

  /** state buscarListasStatus — status of the request buscarListas */
  @property({ attribute: false }) buscarListasStatus: RequestStatus = 'idle';

  /** state buscarListasError — error of the request buscarListas; null when none */
  @property({ attribute: false }) buscarListasError: ErrorState = null;

  /** state criarListaStatus — status of the command criarLista; stays on success until the person changes something */
  @property({ attribute: false }) criarListaStatus: RequestStatus = 'idle';

  /** state criarListaError — error of the command criarLista; null when none */
  @property({ attribute: false }) criarListaError: ErrorState = null;

  /** state salvarListaStatus — status of the command salvarLista; stays on success until the person changes something */
  @property({ attribute: false }) salvarListaStatus: RequestStatus = 'idle';

  /** state salvarListaError — error of the command salvarLista; null when none */
  @property({ attribute: false }) salvarListaError: ErrorState = null;

  /** state scenary — the visible scene of the page; '' until the page shows its first scene */
  @property({ attribute: false }) scenary = '';

  private publish<M extends StateMember>(member: M, value: this[M]): void {
    (this as unknown as Record<string, unknown>)[member] = value;
    setState(`ui.tarefas.listas.${member}`, value);
    if (member === 'selectedLista') {
      try {
        if (value === null || value === undefined) {
          localStorage.removeItem(LOCAL_STORAGE_LISTA_ID);
        } else {
          localStorage.setItem(LOCAL_STORAGE_LISTA_ID, String(value));
        }
      } catch {
        // storage unavailable: the selection simply is not persisted
      }
    }
  }

  private assignState(key: string, value: unknown): void {
    const member = STATE_MEMBER_BY_KEY[key];
    if (member) (this as unknown as Record<string, unknown>)[member] = value;
  }

  public connectedCallback(): void {
    super.connectedCallback();
    const storedListas = getState('ui.tarefas.listas.listas');
    if (storedListas !== undefined) this.assignState('ui.tarefas.listas.listas', storedListas);
    this.resetVisit();
    subscribe([...STATE_KEYS], this);
    this.applyEntryParams();
    void this.listarListas();
  }

  public disconnectedCallback(): void {
    unsubscribe([...STATE_KEYS], this);
    super.disconnectedCallback();
  }

  public handleIcaStateChange(key: string, value: any): void {
    if (!(key in STATE_MEMBER_BY_KEY)) {
      super.handleIcaStateChange(key, value);
      return;
    }
    if (value === undefined) return;
    this.assignState(key, value);
    this.requestUpdate();
  }

  private resetVisit(): void {
    this.publish('pageStatus', 'idle');
    this.publish('listarListasStatus', 'idle');
    this.publish('listarListasError', null);
    this.publish('buscarListasStatus', 'idle');
    this.publish('buscarListasError', null);
    this.publish('criarListaStatus', 'idle');
    this.publish('criarListaError', null);
    this.publish('salvarListaStatus', 'idle');
    this.publish('salvarListaError', null);
    this.publish('scenary', '');
    this.publish('busca', '');
    this.publish('criarListaDraft', emptyCriarListaDraft());
    this.publish('salvarListaDraft', emptySalvarListaDraft());
    this.publish('selectedLista', null);
  }

  private resetCommandResults(): void {
    this.publish('criarListaStatus', 'idle');
    this.publish('criarListaError', null);
    this.publish('salvarListaStatus', 'idle');
    this.publish('salvarListaError', null);
  }

  private applyEntryParams(): void {
    let raw: string | null = new URLSearchParams(window.location.search).get('listaId');
    if (raw === null) {
      try {
        raw = localStorage.getItem(LOCAL_STORAGE_LISTA_ID);
      } catch {
        raw = null;
      }
    }
    if (raw !== null && raw !== '') {
      // the row is resolved once the list is loaded
      this.publish('selectedLista', raw);
    }
  }

  private unexpectedError(error: unknown): ErrorState {
    const name = error instanceof Error ? error.name : 'unknown';
    return { code: 'client.unexpected', message: '', details: { name } };
  }

  private upsertLista(lista: ListaCadastrada): void {
    const exists = this.listas.some(row => row.id === lista.id);
    const next = exists ? this.listas.map(row => (row.id === lista.id ? lista : row)) : [...this.listas, lista];
    this.publish('listas', next);
  }

  /** The row of the loaded lists whose id is selectedLista (the selected record of formularioLista); null when nothing is selected or the row is not loaded */
  public get formularioLista(): ListaCadastrada | null {
    const id = this.selectedLista;
    if (id === null) return null;
    const row = this.listas.find(item => item.id === id);
    return row !== undefined ? row : null;
  }

  /** select — selects the list (by id) that formularioLista edits, or clears the selection with null; fills salvarListaDraft from the row; an id with no loaded row publishes client.selectionInvalid in listarListasError; id: the list id or null */
  public selectFormularioLista(id: string | null): void {
    this.resetCommandResults();
    this.publish('selectedLista', id);
    this.applySelection();
  }

  private applySelection(): void {
    const id = this.selectedLista;
    if (id === null) {
      this.publish('salvarListaDraft', emptySalvarListaDraft());
      return;
    }
    const row = this.listas.find(item => item.id === id);
    if (row !== undefined) {
      this.publish('salvarListaDraft', draftFromLista(row));
      return;
    }
    if (this.listarListasStatus === 'success' || this.buscarListasStatus === 'success') {
      this.publish('salvarListaDraft', emptySalvarListaDraft());
      this.publish('listarListasError', {
        code: 'client.selectionInvalid',
        message: '',
        details: { listaId: id },
      });
    }
  }

  /** draft — replaces the draft of the new-list form (criarLista); value: the whole draft with nome, details.tipo, details.descricao and ativa */
  public setCriarListaDraft(value: CriarListaDraft): void {
    this.resetCommandResults();
    this.publish('criarListaDraft', value);
  }

  /** draft — replaces the draft of the edit-list form (salvarLista); value: the whole draft with id, version, nome, details and ativa */
  public setSalvarListaDraft(value: SalvarListaDraft): void {
    this.resetCommandResults();
    this.publish('salvarListaDraft', value);
  }

  /** Publishes the visible scene of the page; value: the scene id ('' for none) */
  public setScenario(value: string): void {
    this.resetCommandResults();
    this.publish('scenary', value);
  }

  /** request listarListas — loads every list of the organization (active and inactive, ordered by name, with open-task count); redraws listas (replace), then resolves the selected list and sets pageStatus; no arguments; also the method to reload the page */
  public async listarListas(): Promise<void> {
    if (this.listarListasStatus === 'loading') return;
    this.publish('listarListasStatus', 'loading');
    this.publish('listarListasError', null);
    this.publish('pageStatus', 'loading');
    try {
      const input: Input<'tarefas.listas.listarListas'> = {};
      const response = await execBff<Output<'tarefas.listas.listarListas'>>('tarefas.listas.listarListas', input, { mode: 'silent' });
      if (!response.ok || !response.data) {
        this.publish('listarListasError', response.error ?? this.unexpectedError(null));
        this.publish('listarListasStatus', 'error');
        this.publish('pageStatus', 'error');
        return;
      }
      const listas = response.data.listas;
      this.publish('listas', listas);
      this.publish('listarListasStatus', 'success');
      this.applySelection();
      this.publish('pageStatus', listas.length === 0 ? 'empty' : 'success');
    } catch (error) {
      this.publish('listarListasError', this.unexpectedError(error));
      this.publish('listarListasStatus', 'error');
      this.publish('pageStatus', 'error');
    }
  }

  /** request buscarListas — searches the lists whose name contains the typed fragment (case-insensitive, ordered by name); redraws listas (replace) and keeps the fragment in busca; busca: the name fragment, '' lists all */
  public async buscarListas(busca: string): Promise<void> {
    if (this.buscarListasStatus === 'loading') return;
    this.publish('busca', busca);
    this.publish('buscarListasStatus', 'loading');
    this.publish('buscarListasError', null);
    try {
      const input: Input<'tarefas.listas.buscarListas'> = { busca };
      const response = await execBff<Output<'tarefas.listas.buscarListas'>>('tarefas.listas.buscarListas', input, { mode: 'silent' });
      if (!response.ok || !response.data) {
        this.publish('buscarListasError', response.error ?? this.unexpectedError(null));
        this.publish('buscarListasStatus', 'error');
        return;
      }
      this.publish('listas', response.data.listas);
      this.publish('buscarListasStatus', 'success');
    } catch (error) {
      this.publish('buscarListasError', this.unexpectedError(error));
      this.publish('buscarListasStatus', 'error');
    }
  }

  /** request criarLista — creates a new list (project or area) from criarListaDraft; redraws listas (upsert of the returned list by id) and clears criarListaDraft; no arguments, the person fills the draft with setCriarListaDraft */
  public async criarLista(): Promise<void> {
    const draft = this.criarListaDraft;
    const missing: string[] = [];
    if (draft.nome === null || draft.nome === '') missing.push('criarListaDraft.nome');
    if (draft.details.tipo === null) missing.push('criarListaDraft.details.tipo');
    if (draft.ativa === null) missing.push('criarListaDraft.ativa');
    if (draft.nome === null || draft.details.tipo === null || draft.ativa === null || missing.length > 0) {
      this.publish('criarListaError', { code: 'client.requiredMissing', message: '', details: { stateKeys: missing } });
      this.publish('criarListaStatus', 'error');
      return;
    }
    const input: Input<'tarefas.listas.criarLista'> = {
      nome: draft.nome,
      details: buildDetails(draft.details, draft.details.tipo),
      ativa: draft.ativa,
    };
    this.publish('criarListaStatus', 'loading');
    this.publish('criarListaError', null);
    try {
      const response = await runBlockingUiAction(signal =>
        execBff<Output<'tarefas.listas.criarLista'>>('tarefas.listas.criarLista', input, { mode: 'blocking', signal }),
      );
      if (response === undefined) {
        this.publish('criarListaError', { code: 'client.unexpected', message: '', details: { name: 'aborted' } });
        this.publish('criarListaStatus', 'error');
        return;
      }
      if (!response.ok || !response.data) {
        this.publish('criarListaError', response.error ?? this.unexpectedError(null));
        this.publish('criarListaStatus', 'error');
        return;
      }
      this.upsertLista(response.data.lista);
      this.publish('criarListaDraft', emptyCriarListaDraft());
      this.publish('criarListaStatus', 'success');
    } catch (error) {
      this.publish('criarListaError', this.unexpectedError(error));
      this.publish('criarListaStatus', 'error');
    }
  }

  /** request salvarLista — saves the edits of the selected list from salvarListaDraft (id and version come from the draft, filled by selectFormularioLista); redraws listas (upsert of the returned list by id) and refills salvarListaDraft from the saved list with its new version; no arguments */
  public async salvarLista(): Promise<void> {
    const draft = this.salvarListaDraft;
    const missing: string[] = [];
    if (draft.id === null || draft.id === '') missing.push('salvarListaDraft.id');
    if (draft.version === null) missing.push('salvarListaDraft.version');
    if (draft.nome === null || draft.nome === '') missing.push('salvarListaDraft.nome');
    if (draft.details.tipo === null) missing.push('salvarListaDraft.details.tipo');
    if (draft.ativa === null) missing.push('salvarListaDraft.ativa');
    if (
      draft.id === null ||
      draft.version === null ||
      draft.nome === null ||
      draft.details.tipo === null ||
      draft.ativa === null ||
      missing.length > 0
    ) {
      this.publish('salvarListaError', { code: 'client.requiredMissing', message: '', details: { stateKeys: missing } });
      this.publish('salvarListaStatus', 'error');
      return;
    }
    const input: Input<'tarefas.listas.salvarLista'> = {
      id: draft.id,
      version: draft.version,
      nome: draft.nome,
      details: buildDetails(draft.details, draft.details.tipo),
      ativa: draft.ativa,
    };
    this.publish('salvarListaStatus', 'loading');
    this.publish('salvarListaError', null);
    try {
      const response = await runBlockingUiAction(signal =>
        execBff<Output<'tarefas.listas.salvarLista'>>('tarefas.listas.salvarLista', input, { mode: 'blocking', signal }),
      );
      if (response === undefined) {
        this.publish('salvarListaError', { code: 'client.unexpected', message: '', details: { name: 'aborted' } });
        this.publish('salvarListaStatus', 'error');
        return;
      }
      if (!response.ok || !response.data) {
        this.publish('salvarListaError', response.error ?? this.unexpectedError(null));
        this.publish('salvarListaStatus', 'error');
        return;
      }
      const lista = response.data.lista;
      this.upsertLista(lista);
      this.publish('salvarListaDraft', draftFromLista(lista));
      this.publish('salvarListaStatus', 'success');
    } catch (error) {
      this.publish('salvarListaError', this.unexpectedError(error));
      this.publish('salvarListaStatus', 'error');
    }
  }
}
