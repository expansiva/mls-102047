type Status = TarefaQuadroItem['status'];
type Prioridade = TarefaQuadroItem['prioridade'];
type ModalHost = HTMLElement & { visible: boolean };

const lang = () => document.documentElement.lang || undefined;

function fmtDate(v: string): string {
  const d = new Date(v);
  if (Number.isNaN(d.getTime())) return v;
  return new Intl.DateTimeFormat(lang(), { dateStyle: 'short', timeZone: v.length === 10 ? 'UTC' : undefined }).format(d);
}

function fmtDateTime(v: string): string {
  const d = new Date(v);
  if (Number.isNaN(d.getTime())) return v;
  return new Intl.DateTimeFormat(lang(), { dateStyle: 'short', timeStyle: 'short' }).format(d);
}

function statusLabel(m: PageMessageType, s: Status): string {
  switch (s) {
    case 'aFazer': return m.stAFazer;
    case 'emAndamento': return m.stEmAndamento;
    case 'emRevisao': return m.stEmRevisao;
    case 'concluida': return m.stConcluida;
    default: return m.stCancelada;
  }
}

function statusClass(s: Status): string {
  switch (s) {
    case 'emAndamento': return 'bg-[var(--status-info-bg,transparent)] text-[var(--status-info-text,currentColor)]';
    case 'emRevisao': return 'bg-[var(--status-warning-bg,transparent)] text-[var(--status-warning-text,currentColor)]';
    case 'concluida': return 'bg-[var(--status-success-bg,transparent)] text-[var(--status-success-text,currentColor)]';
    default: return 'bg-[var(--status-neutral-bg,transparent)] text-[var(--status-neutral-text,currentColor)]';
  }
}

function prioridadeLabel(m: PageMessageType, p: Prioridade): string {
  switch (p) {
    case 'high': return m.prHigh;
    case 'medium': return m.prMedium;
    default: return m.prLow;
  }
}

function chip(cls: string, text: string) {
  return html`<span class="inline-flex items-center whitespace-nowrap rounded-full px-2 py-0.5 text-xs font-medium ${cls}">${text}</span>`;
}

function nameOf(m: MembroResumo): string {
  return m.details.identification.details.identification.name;
}

function setModal(e: Event, visible: boolean) {
  const host = e.currentTarget as HTMLElement;
  const root = host.closest('[data-organism-id="acoesGestorTarefa"]');
  const modal = root?.querySelector('groupnotifyuser--ml-alert-modal') as ModalHost | null;
  if (modal) modal.visible = visible;
}

          </Action>
        </groupnotifyuser--ml-contextual-feedback>
        <groupnotifyuser--ml-contextual-feedback type="error" .visible=${this.cmdCancelarTarefaStatus === 'error'} .dismissible=${false}>
          <Message>${m.cancelFail}</Message>
          <Action>
            <grouptriggeraction--ml-button-standard data-variant="secondary" size="xs" @action=${() => this.cmdCancelarTarefa()}>
              <Label>${m.retry}</Label>
            </grouptriggeraction--ml-button-standard>
          </Action>
        </groupnotifyuser--ml-contextual-feedback>`
      : nothing;
    let controls;
    if (!t) {
      controls = nothing;
    } else if (!t.details.details.aberta) {
      controls = html`<p class="text-sm text-[var(--text-muted,currentColor)]">${m.closedNote}</p>`;
    } else {
      controls = html`
        <div class="flex flex-col gap-3 rounded-md border border-[var(--border-subtle,currentColor)] bg-[var(--surface-alt-bg,transparent)] p-3">
          <groupselectone--ml-select-one-autocomplete
            .value=${draft.responsavelId}
            clearable
            placeholder=${m.newResponsavelPlaceholder}
            @focus=${() => this.qryBuscarMembros(null, true)}
            @input=${(e: Event) => {
              const v = (e as CustomEvent<{ value?: string }>).detail?.value;
              if (typeof v === 'string') this.qryBuscarMembros(v, true);
            }}
            @change=${(e: CustomEvent<{ value: string | null }>) => this.setCmdReatribuirTarefaDraft({ responsavelId: e.detail.value })}>
            <Label>${m.newResponsavel}</Label>
            ${this.membros.map((mb) => html`<Item value=${mb.id}>${nameOf(mb)}</Item>`)}
            <Empty>${m.noMember}</Empty>
          </groupselectone--ml-select-one-autocomplete>
          <div class="flex flex-wrap items-center justify-between gap-2">
            <grouptriggeraction--ml-button-standard data-variant="primary"
              .disabled=${!draft.responsavelId || busy}
              .loading=${this.cmdReatribuirTarefaStatus === 'loading'}
              @action=${() => this.cmdReatribuirTarefa()}>
              <Label>${m.reassign}</Label>
            </grouptriggeraction--ml-button-standard>
            <grouptriggeraction--ml-button-standard data-variant="danger"
              .disabled=${busy}
              .loading=${this.cmdCancelarTarefaStatus === 'loading'}
              @action=${(e: Event) => setModal(e, true)}>
              <Label>${m.cancelTask}</Label>
            </grouptriggeraction--ml-button-standard>
          </div>
        </div>
        <groupnotifyuser--ml-alert-modal type="danger" position="center" .dismissible=${true}
          @dismiss=${(e: Event) => setModal(e, false)}>
          <Title>${m.confirmTitle}</Title>
          <Message>
            <p>${m.confirmBody}</p>
            <p class="mt-2 font-semibold">${t.titulo}</p>
          </Message>
          <Action>
            <div class="flex gap-2">
              <grouptriggeraction--ml-button-standard data-variant="secondary" @action=${(e: Event) => setModal(e, false)}>
                <Label>${m.keepTask}</Label>
              </grouptriggeraction--ml-button-standard>
              <grouptriggeraction--ml-button-standard data-variant="danger"
                @action=${(e: Event) => { setModal(e, false); this.cmdCancelarTarefa(); }}>
                <Label>${m.confirmCancel}</Label>
              </grouptriggeraction--ml-button-standard>
            </div>
          </Action>
        </groupnotifyuser--ml-alert-modal>`;
    }
    return html`<section data-organism-id="acoesGestorTarefa" class="flex flex-col gap-3" aria-live="polite">${feedback}${controls}</section>`;
  }

  private renderHistorico() {
    const m = this.msg;
    const t = this.qryDetalheTarefaStatus === 'loading' || this.qryDetalheTarefaError ? null : this.tarefa;
    if (!t) {
      return html`<section data-organism-id="historicoTarefa"><p class="py-6 text-sm text-[var(--text-muted,currentColor)]">${m.historyEmpty}</p></section>`;
    }
    const d = t.details.details;
    return html`
      <section data-organism-id="historicoTarefa">
        <groupviewdata--ml-timeline-view>
          <Columns>
            <Column field="quando" header=${m.colQuando}></Column>
            <Column field="evento" header=${m.colEvento}></Column>
          </Columns>
          <Rows>
            <Row>
              <Cell>${fmtDateTime(t.criadaEm)}</Cell>
              <Cell>
                <div class="font-medium text-[var(--text-strong,currentColor)]">${m.evCriada}</div>
                <div class="text-xs text-[var(--text-muted,currentColor)]">${m.evCriadaPor} ${nameOf(t.criador)}</div>
              </Cell>
            </Row>
            ${d.devolvidaEm
              ? html`<Row>
                  <Cell>${fmtDateTime(d.devolvidaEm)}</Cell>
                  <Cell>
                    <div class="font-medium text-[var(--text-strong,currentColor)]">${m.evDevolvida}</div>
                    ${d.motivoDevolucao ? html`<div class="text-xs text-[var(--text-muted,currentColor)]">${m.evReason}: ${d.motivoDevolucao}</div>` : nothing}
                  </Cell>
                </Row>`
              : nothing}
            ${d.concluidaEm
              ? html`<Row>
                  <Cell>${fmtDateTime(d.concluidaEm)}</Cell>
                  <Cell><div class="font-medium text-[var(--text-strong,currentColor)]">${m.evConcluida}</div></Cell>
                </Row>`
              : nothing}
            ${d.canceladaEm
              ? html`<Row>
                  <Cell>${fmtDateTime(d.canceladaEm)}</Cell>
                  <Cell><div class="font-medium text-[var(--text-strong,currentColor)]">${m.evCancelada}</div></Cell>
                </Row>`
              : nothing}
          </Rows>
        </groupviewdata--ml-timeline-view>
      </section>`;
  }
