type PageMessageType = typeof pageMessage_pt_br;
const pageMessages: Record<string, PageMessageType> = { 'pt-br': pageMessage_pt_br };

function isDirty(rec: PessoaEquipe | null, draft: CmdSalvarAlteracoesDraft): boolean {
  if (!rec) return false;
  const a = rec.details.identification;
  const b = draft.details.identification;
  return (b.name ?? '') !== a.name || (b.countryCode ?? '') !== a.countryCode;
}

@customElement('tarefas--web--desktop--page11--membros_equipe-102047')
export class TarefasDesktopPage11Membros_equipePage extends TarefasMembros_equipeShared {
  private msg: PageMessageType = pageMessage_pt_br;

  private statusLabel(s: 'Active' | 'Inactive' | 'Merged' | 'Blocked') {
    switch (s) {
      case 'Active': return this.msg.statusActive;
      case 'Inactive': return this.msg.statusInactive;
      case 'Merged': return this.msg.statusMerged;
      default: return this.msg.statusBlocked;
    }
  }

  private selectRow(index: number) {
    const item = this.pessoas?.items[index];
    if (!item) return;
    if (item.id === this.selectedMembroEquipe && this.scenary === 'editar') return;
    if (this.scenary === 'editar' && isDirty(this.cadastroPessoa, this.cmdSalvarAlteracoesDraft)) {
      if (!window.confirm(this.msg.discardConfirm)) return;
    }
    this.selectMembroEquipe(item.id);
    this.setScenario('editar');
  }

  private renderFeedback(type: 'error' | 'success', visible: boolean, text: string) {
    return html`<groupnotifyuser--ml-contextual-feedback type=${type} ?visible=${visible} .dismissible=${false}>
      <Message>${text}</Message>
    </groupnotifyuser--ml-contextual-feedback>`;
  }

  private renderTable() {
    const items = this.pessoas?.items ?? [];
    const selIdx = items.findIndex((p) => p.id === this.selectedMembroEquipe);
    const loading = this.pageStatus === 'loading' || this.qryPessoasEquipeStatus === 'loading';
    return html`
      <groupviewtable--ml-data-table
        .value=${selIdx >= 0 ? String(selIdx) : ''}
        ?loading=${loading && items.length === 0}
        @rowClick=${(e: CustomEvent<{ index: number }>) => this.selectRow(e.detail.index)}
      >
        <TableHeader>
          <TableRow>
            <TableHead key="name" sortable>${this.msg.colName}</TableHead>
            <TableHead key="country">${this.msg.colCountry}</TableHead>
            <TableHead key="status">${this.msg.colStatus}</TableHead>
            <TableHead key="contacts">${this.msg.colContacts}</TableHead>
          </TableRow>
        </TableHeader>
        <TableBody>
          ${items.map((p) => {
            const id = p.details.identification;
            const muted = id.status !== 'Active';
            const tone = muted
              ? 'text-[var(--text-muted,currentColor)]'
              : 'text-[var(--text-strong,currentColor)]';
            return html`<TableRow key=${p.id}>
              <TableCell><span class="font-medium ${tone}">${id.name}</span></TableCell>
              <TableCell><span class=${muted ? 'text-[var(--text-muted,currentColor)]' : 'text-[var(--text-default,currentColor)]'}>${id.countryCode}</span></TableCell>
              <TableCell>
                <span class="inline-block rounded-full px-2 py-0.5 text-xs ${muted
                  ? 'bg-[var(--status-neutral-bg,transparent)] text-[var(--status-neutral-text,currentColor)]'
                  : 'bg-[var(--status-success-bg,transparent)] text-[var(--status-success-text,currentColor)]'}">${this.statusLabel(id.status)}</span>
              </TableCell>
              <TableCell><span class="text-sm text-[var(--text-muted,currentColor)]">${p.details.base.contacts || this.msg.noValue}</span></TableCell>
            </TableRow>`;
          })}
        </TableBody>
        <Empty>${this.busca ? this.msg.emptyListSearch : this.msg.emptyList}</Empty>
      </groupviewtable--ml-data-table>
      ${this.pessoas?.hasMore
        ? html`<div class="mt-3 flex justify-center">
            <grouptriggeraction--ml-button-standard
              data-variant="secondary"
              size="sm"
              ?loading=${this.qryPessoasEquipeStatus === 'loading'}
              @action=${() => this.qryPessoasEquipe(undefined, (this.pessoas?.page ?? 1) + 1, true)}
            ><Label>${this.msg.loadMore}</Label></grouptriggeraction--ml-button-standard>
          </div>`
        : nothing}
    `;
  }

  private renderListError() {
    const err = this.qryPessoasEquipeError;
    if (!err) return nothing;
    if (err.code === 'client.preconditionMissing') {
      return html`<div class="rounded-lg border border-[var(--border-default,currentColor)] bg-[var(--surface-alt-bg,transparent)] p-6" aria-live="polite">
        <p class="font-medium text-[var(--text-strong,currentColor)]">${this.msg.missingTitle}</p>
        <p class="mt-1 text-sm text-[var(--text-default,currentColor)]">${this.msg.missingBody}</p>
        <div class="mt-4">
          <grouptriggeraction--ml-button-standard data-variant="secondary" @action=${() => history.back()}>
            <Label>${this.msg.back}</Label>
          </grouptriggeraction--ml-button-standard>
        </div>
      </div>`;
    }
    return html`<div class="mb-3 flex items-center justify-between gap-3 rounded-lg bg-[var(--status-error-bg,transparent)] px-3 py-2 text-sm text-[var(--status-error-text,currentColor)]" aria-live="polite">
      <span>${this.msg.listError} ${err.message}</span>
      <grouptriggeraction--ml-button-standard data-variant="secondary" size="sm" @action=${() => this.qryPessoasEquipe()}>
        <Label>${this.msg.retry}</Label>
      </grouptriggeraction--ml-button-standard>
    </div>`;
  }

  private renderNovo() {
    const d = this.cmdCadastrarPessoaDraft;
    const id = d.details.identification;
    const loading = this.cmdCadastrarPessoaStatus === 'loading';
    const canSave = !!(id.name && id.name.trim()) && !!(id.countryCode && id.countryCode.trim());
    const justDone = this.cmdCadastrarPessoaStatus === 'success';
    const doneName = this.cmdCadastrarPessoaPessoa?.details.identification.name ?? '';
    return html`<form class="flex flex-col gap-4" @submit=${(e: Event) => e.preventDefault()}>
      <groupentertext--ml-enter-text
        .value=${id.name ?? ''}
        required
        @input=${(e: CustomEvent<{ value: string }>) =>
          this.setCmdCadastrarPessoaDraft({ details: { identification: { ...id, name: e.detail.value } } })}
      >
        <Label>${this.msg.fieldName}</Label>
        <Helper>${this.msg.fieldNameHelper}</Helper>
      </groupentertext--ml-enter-text>
      <groupentertext--ml-enter-text
        .value=${id.countryCode ?? ''}
        placeholder=${this.msg.countryPlaceholder}
        required
        @input=${(e: CustomEvent<{ value: string }>) =>
          this.setCmdCadastrarPessoaDraft({ details: { identification: { ...id, countryCode: e.detail.value } } })}
      >
        <Label>${this.msg.fieldCountry}</Label>
        <Helper>${this.msg.fieldCountryHelper}</Helper>
      </groupentertext--ml-enter-text>
      <div aria-live="polite" class="flex flex-col gap-2">
        ${this.renderFeedback('error', this.cmdCadastrarPessoaStatus === 'error', `${this.msg.registerFailed} ${this.cmdCadastrarPessoaError?.message ?? ''}`)}
        ${this.renderFeedback('success', justDone, `${this.msg.registered} ${doneName}`)}
      </div>
      <grouptriggeraction--ml-button-standard
        data-class="w-full"
        ?disabled=${!canSave}
        ?loading=${loading}
        @action=${() => this.cmdCadastrarPessoa()}
      ><Label>${this.msg.registerPerson}</Label></grouptriggeraction--ml-button-standard>
    </form>`;
  }

  private renderEditar() {
    const rec = this.cadastroPessoa;
    const d = this.cmdSalvarAlteracoesDraft;
    const id = d.details.identification;
    const loading = this.cmdSalvarAlteracoesStatus === 'loading';
    const filled = !!(id.name && id.name.trim()) && !!(id.countryCode && id.countryCode.trim());
    const canSave = filled && isDirty(rec, d);
    const ident = rec?.details.identification;
    return html`<form class="flex flex-col gap-4" @submit=${(e: Event) => e.preventDefault()}>
      <groupentertext--ml-enter-text
        .value=${id.name ?? ''}
        required
        @input=${(e: CustomEvent<{ value: string }>) =>
          this.setCmdSalvarAlteracoesDraft({ ...d, details: { identification: { ...id, name: e.detail.value } } })}
      >
        <Label>${this.msg.fieldName}</Label>
        <Helper>${this.msg.fieldNameHelper}</Helper>
      </groupentertext--ml-enter-text>
      <groupentertext--ml-enter-text
        .value=${id.countryCode ?? ''}
        placeholder=${this.msg.countryPlaceholder}
        required
        @input=${(e: CustomEvent<{ value: string }>) =>
          this.setCmdSalvarAlteracoesDraft({ ...d, details: { identification: { ...id, countryCode: e.detail.value } } })}
      >
        <Label>${this.msg.fieldCountry}</Label>
        <Helper>${this.msg.fieldCountryHelper}</Helper>
      </groupentertext--ml-enter-text>
      ${ident && rec
        ? html`<dl class="grid grid-cols-[auto_1fr] gap-x-4 gap-y-1 border-t border-[var(--border-subtle,currentColor)] pt-3 text-sm">
            <dt class="text-[var(--text-muted,currentColor)]">${this.msg.readOnlyStatus}</dt>
            <dd class="text-[var(--text-default,currentColor)]">${this.statusLabel(ident.status)}${ident.status === 'Active' ? html` · <span class="text-[var(--text-muted,currentColor)]">${this.msg.activeHint}</span>` : nothing}</dd>
            <dt class="text-[var(--text-muted,currentColor)]">${this.msg.readOnlyRoles}</dt>
            <dd class="text-[var(--text-default,currentColor)]">${ident.tags || this.msg.noValue}</dd>
            <dt class="text-[var(--text-muted,currentColor)]">${this.msg.readOnlyContacts}</dt>
            <dd class="text-[var(--text-default,currentColor)]">${rec.details.base.contacts || this.msg.noValue}</dd>
          </dl>`
        : nothing}
      <div aria-live="polite" class="flex flex-col gap-2">
        ${this.renderFeedback('error', this.cmdSalvarAlteracoesStatus === 'error', `${this.msg.saveFailed} ${this.cmdSalvarAlteracoesError?.message ?? ''}`)}
        ${this.renderFeedback('success', this.cmdSalvarAlteracoesStatus === 'success', this.msg.saved)}
      </div>
      <grouptriggeraction--ml-button-standard
        data-class="w-full"
        ?disabled=${!canSave}
        ?loading=${loading}
        @action=${() => this.cmdSalvarAlteracoes()}
      ><Label>${this.msg.saveChanges}</Label></grouptriggeraction--ml-button-standard>
    </form>`;
  }

  private renderPanel() {
    const rec = this.cadastroPessoa;
    const editTitle = rec ? rec.details.identification.name : this.msg.editTitleFallback;
    return html`<molecules--ml-scenary-102020
      mode="scenary"
      .value=${this.scenary || 'inicio'}
      backLabel=${this.msg.closePanel}
      @change=${(e: CustomEvent<{ value: string }>) => {
        if (e.target === e.currentTarget) {
          if (e.detail.value === 'inicio') this.selectMembroEquipe(null);
          this.setScenario(e.detail.value);
        }
      }}
    >
      <Scene value="inicio" title=${this.msg.idleTitle}>
        <p class="text-sm text-[var(--text-muted,currentColor)]">${this.msg.idleHint}</p>
      </Scene>
      <Scene value="novo" title=${this.msg.newTitle} nav="back" backTo="inicio">
        ${this.renderNovo()}
      </Scene>
      <Scene value="editar" title=${editTitle} nav="back" backTo="inicio">
        ${this.renderEditar()}
      </Scene>
    </molecules--ml-scenary-102020>`;
  }

  render() {
    this.msg = pageMessages[this.getMessageKey(pageMessages)];
    const missing = this.qryPessoasEquipeError?.code === 'client.preconditionMissing';
    return html`
      <div class="min-h-full bg-[var(--page-bg,transparent)] p-6 text-[var(--text-default,currentColor)]">
        <header class="mb-5">
          <h1 class="text-2xl font-semibold text-[var(--text-strong,currentColor)]">${this.msg.pageTitle}</h1>
          <p class="text-sm text-[var(--text-muted,currentColor)]">${this.msg.pageSubtitle}</p>
        </header>
        ${missing
          ? this.renderListError()
          : html`<div class="grid grid-cols-[minmax(0,1fr)_22rem] items-start gap-6">
              <section data-organism-id="pessoasEquipe" class="min-w-0">
                <div class="mb-3 flex items-end justify-between gap-4">
                  <div class="max-w-sm flex-1">
                    <groupsearchcontent--ml-search-bar
                      .value=${this.busca}
                      placeholder=${this.msg.searchPlaceholder}
                      debounce="300"
                      ?loading=${this.qryPessoasEquipeStatus === 'loading'}
                      @search=${(e: CustomEvent<{ query: string }>) => this.qryPessoasEquipe(e.detail.query)}
                      @clear=${() => this.qryPessoasEquipe(null)}
                    >
                      <Label data-class="sr-only">${this.msg.searchLabel}</Label>
                    </groupsearchcontent--ml-search-bar>
                  </div>
                  <grouptriggeraction--ml-button-standard
                    @action=${() => {
                      this.selectMembroEquipe(null);
                      this.setScenario('novo');
                    }}
                  ><Label>${this.msg.newPerson}</Label></grouptriggeraction--ml-button-standard>
                </div>
                ${this.renderListError()}
                <div class="rounded-lg border border-[var(--border-default,currentColor)] bg-[var(--surface-bg,transparent)] p-2" aria-busy=${this.qryPessoasEquipeStatus === 'loading' ? 'true' : 'false'}>
                  ${this.renderTable()}
                </div>
              </section>
              <aside data-organism-id="cadastroPessoa" class="sticky top-6 rounded-lg border border-[var(--border-default,currentColor)] bg-[var(--surface-bg,transparent)] p-4">
                ${this.renderPanel()}
              </aside>
            </div>`}
      </div>
    `;
  }
}
