function fmtDate(v: string | null | undefined): string {
  if (!v) return '';
  const d = new Date(v.length === 10 ? v + 'T00:00:00' : v);
  if (Number.isNaN(d.getTime())) return v;
  return new Intl.DateTimeFormat(document.documentElement.lang || undefined, { dateStyle: 'medium' }).format(d);
}

function fmtDateTime(v: string | null | undefined): string {
  if (!v) return '';
  const d = new Date(v);
  if (Number.isNaN(d.getTime())) return v;
  return new Intl.DateTimeFormat(document.documentElement.lang || undefined, { dateStyle: 'short', timeStyle: 'short' }).format(d);
}

function setModal(e: Event, visible: boolean) {
  const el = (e.currentTarget as HTMLElement)
    .closest('[data-organism-id="acoesGestor"]')
    ?.querySelector<HTMLElement & { visible: boolean }>('[data-modal="cancelar"]');
  if (el) el.visible = visible;
}

                ? html`<p role="alert" aria-live="polite" class="text-sm text-[var(--text-default,currentColor)]">${m.erroReatribuir}</p>`
                : nothing}
              <grouptriggeraction--ml-button-standard
                data-variant="primary"
                size="lg"
                data-class="w-full"
                ?loading=${reLoading}
                ?disabled=${this.reatribuirTarefaDraft.responsavelId === null}
                @action=${() => { void this.reatribuirTarefa(); }}
              >
                <Label>${m.reatribuir}</Label>
              </grouptriggeraction--ml-button-standard>
              <hr class="border-[var(--border-subtle,currentColor)]" />
              ${this.cancelarTarefaError
                ? html`<p role="alert" aria-live="polite" class="text-sm text-[var(--text-default,currentColor)]">${m.erroCancelar}</p>`
                : nothing}
              <grouptriggeraction--ml-button-standard
                data-variant="danger"
                size="lg"
                data-class="w-full"
                ?loading=${caLoading}
                @action=${(e: Event) => setModal(e, true)}
              >
                <Label>${m.cancelar}</Label>
              </grouptriggeraction--ml-button-standard>
              <groupnotifyuser--ml-alert-modal data-modal="cancelar" type="warning" .dismissible=${false}>
                <Title>${m.confirmaTitulo}</Title>
                <Message>${m.confirmaMsg} <strong>${t.titulo}</strong>. ${m.confirmaAviso}</Message>
                <Action>
                  <div class="flex w-full gap-2">
                    <grouptriggeraction--ml-button-standard data-variant="secondary" size="lg" data-class="flex-1" @action=${(e: Event) => setModal(e, false)}>
                      <Label>${m.manter}</Label>
                    </grouptriggeraction--ml-button-standard>
                    <grouptriggeraction--ml-button-standard
                      data-variant="danger"
                      size="lg"
                      data-class="flex-1"
                      @action=${(e: Event) => { setModal(e, false); void this.cancelarTarefa(); }}
                    >
                      <Label>${m.confirmarCancelar}</Label>
                    </grouptriggeraction--ml-button-standard>
                  </div>
                </Action>
              </groupnotifyuser--ml-alert-modal>
            `
          : html`<p class="text-sm text-[var(--text-muted,currentColor)]">${m.encerrada}</p>`}
        <groupnotifyuser--ml-toast-notification type="success" position="bottom-right" .duration=${4000} .visible=${this.reatribuirTarefaStatus === 'success'}>
          <Message>${m.toastReatribuida}</Message>
        </groupnotifyuser--ml-toast-notification>
        <groupnotifyuser--ml-toast-notification type="success" position="bottom-right" .duration=${4000} .visible=${this.cancelarTarefaStatus === 'success'}>
          <Message>${m.toastCancelada}</Message>
        </groupnotifyuser--ml-toast-notification>
      </section>
    `;
  }
