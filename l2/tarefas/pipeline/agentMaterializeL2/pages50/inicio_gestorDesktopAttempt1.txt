function formatDate(value: string): string {
  if (!value) return '';
  const d = value.length === 10 ? new Date(value + 'T00:00:00') : new Date(value);
  if (Number.isNaN(d.getTime())) return value;
  return new Intl.DateTimeFormat(document.documentElement.lang || undefined, { day: '2-digit', month: 'short', year: 'numeric' }).format(d);
}

function formatNumber(n: number): string {
  return new Intl.NumberFormat(document.documentElement.lang || undefined).format(n);
}

@customElement('tarefas--web--desktop--page11--inicio_gestor-102047')
export class TarefasDesktopPage11Inicio_gestorPage extends TarefasInicio_gestorShared {
  private msg: PageMessageType = pageMessage_pt_br;

  private statusLabel(s: string) {
    const m = this.msg;
    const map: Record<string, string> = {
      aFazer: m.statusAFazer,
      emAndamento: m.statusEmAndamento,
      emRevisao: m.statusEmRevisao,
      concluida: m.statusConcluida,
      cancelada: m.statusCancelada
    };
    return map[s] ?? '';
  }

  private prioLabel(p: string) {
    const m = this.msg;
    const map: Record<string, string> = { low: m.prioLow, medium: m.prioMedium, high: m.prioHigh };
    return map[p] ?? '';
  }

  private renderSkeleton(rows: number) {
    return html`<div class="animate-pulse space-y-3" aria-hidden="true">
      ${Array.from({ length: rows }).map(() => html`<div class="h-12 rounded-md bg-[var(--surface-alt-bg,transparent)]"></div>`)}
    </div>`;
  }

  private renderHeadline() {
    const m = this.msg;
    const t = this.totais;
    if (this.pageStatus === 'loading' || this.pageStatus === 'idle') {
      return html`<div class="h-9 w-2/3 animate-pulse rounded-md bg-[var(--surface-alt-bg,transparent)]" aria-hidden="true"></div>`;
    }
    if (this.pageStatus === 'empty') {
      return html`<div aria-live="polite">
        <p class="text-2xl font-semibold text-[var(--text-strong,currentColor)]">${m.goodNews}</p>
        <p class="mt-1 text-sm text-[var(--text-muted,currentColor)]">${m.goodNewsHelp}</p>
      </div>`;
    }
    if (!t) return nothing;
    return html`<p class="text-2xl text-[var(--text-strong,currentColor)]" aria-live="polite">
      <span class="font-semibold tabular-nums">${formatNumber(t.atrasadas)}</span> ${m.hlLate},
      <span class="font-semibold tabular-nums">${formatNumber(t.vencemHoje)}</span> ${m.hlToday}
      <span class="text-[var(--text-muted,currentColor)]">— ${m.hlTail}</span>
    </p>`;
  }

  private renderLane() {
    const m = this.msg;
    const err = this.qryTarefasPedemAcaoError;
    const items = this.tarefas?.items ?? [];
    const loading = this.qryTarefasPedemAcaoStatus === 'loading';

    if (err && err.code === 'client.preconditionMissing') {
      return html`<div class="rounded-lg border border-[var(--border-default,currentColor)] bg-[var(--surface-bg,transparent)] p-5" role="alert">
        <p class="font-medium text-[var(--text-strong,currentColor)]">${m.missingTitle}</p>
        <p class="mt-1 text-sm text-[var(--text-muted,currentColor)]">${m.missingText}</p>
        <div class="mt-3">
          <grouptriggeraction--ml-button-standard data-variant="secondary" size="md" @action=${() => this.abrirTarefasEquipe(null)}>
            <Label>${m.backToTeam}</Label>
          </grouptriggeraction--ml-button-standard>
        </div>
      </div>`;
    }

    if (this.qryTarefasPedemAcaoStatus === 'error') {
      return html`<div class="rounded-lg border border-[var(--border-default,currentColor)] bg-[var(--status-error-bg,transparent)] p-5" role="alert">
        <p class="font-medium text-[var(--status-error-text,currentColor)]">${m.laneErrorTitle}</p>
        ${err?.message ? html`<p class="mt-1 text-sm text-[var(--text-default,currentColor)]">${err.message}</p>` : nothing}
        <div class="mt-3">
          <grouptriggeraction--ml-button-standard data-variant="secondary" size="md" @action=${() => this.qryTarefasPedemAcao(1)}>
            <Label>${m.retry}</Label>
          </grouptriggeraction--ml-button-standard>
        </div>
      </div>`;
    }

    if (loading && items.length === 0) return this.renderSkeleton(4);

    if (this.pageStatus === 'empty') return nothing;

    return html`
      <groupviewdata--ml-vertical-record-list
        .hoverable=${true}
        .loading=${false}
        @row-click=${(e: CustomEvent<{ index: number }>) => {
          const t = items[e.detail.index];
          if (t) this.agirNaTarefa(t.id);
        }}
      >
        <Columns>
          <Column field="tarefa" header=${m.colTask} />
          <Column field="situacao" header=${m.colStatus} width="140px" />
          <Column field="prioridade" header=${m.colPriority} width="110px" />
          <Column field="prazo" header=${m.colDue} width="170px" align="right" />
          <Column field="acao" header=${m.colAction} width="140px" align="right" />
        </Columns>
        <Rows>
          ${items.map((t) => {
            const flags = t.details?.details;
            const decisive = flags?.atrasada
              ? 'text-[var(--status-error-text,currentColor)]'
              : flags?.venceHoje
                ? 'text-[var(--status-warning-text,currentColor)]'
                : 'text-[var(--text-default,currentColor)]';
            const flagText = flags?.atrasada ? m.late : flags?.venceHoje ? m.dueToday : '';
            const who = t.membroEquipe?.details?.identification?.details?.identification?.name ?? '';
            return html`<Row>
              <Cell>
                <div class="font-medium text-[var(--text-strong,currentColor)]">${t.titulo}</div>
                <div class="text-sm text-[var(--text-muted,currentColor)]">
                  ${t.lista?.nome ?? ''}${who ? html` · ${m.by}: ${who}` : nothing}${flags?.progressoChecklist ? html` · ${m.checklist}: ${flags.progressoChecklist}` : nothing}
                </div>
              </Cell>
              <Cell><span class="text-sm text-[var(--text-default,currentColor)]">${this.statusLabel(t.status)}</span></Cell>
              <Cell><span class="text-sm text-[var(--text-muted,currentColor)]">${this.prioLabel(t.prioridade)}</span></Cell>
              <Cell>
                <div class="tabular-nums font-semibold ${decisive}">${formatDate(t.prazo)}</div>
                ${flagText ? html`<div class="text-xs font-medium ${decisive}">${flagText}</div>` : nothing}
              </Cell>
              <Cell><span class="text-sm font-medium text-[var(--link-text,currentColor)]">${m.openTask} →</span></Cell>
            </Row>`;
          })}
        </Rows>
        <Empty><div class="py-6 text-center text-sm text-[var(--text-muted,currentColor)]">${m.emptyLane}</div></Empty>
      </groupviewdata--ml-vertical-record-list>
      ${this.tarefas?.hasMore
        ? html`<div class="mt-3 flex justify-center">
            <grouptriggeraction--ml-button-standard
              data-variant="secondary"
              size="md"
              ?loading=${loading}
              @action=${() => this.qryTarefasPedemAcao((this.tarefas?.page ?? 1) + 1, true)}
            >
              <Label>${m.loadMore}</Label>
            </grouptriggeraction--ml-button-standard>
          </div>`
        : nothing}
    `;
  }

  private renderStrip() {
    const m = this.msg;
    const t = this.totais;
    const loading = this.qryPanoramaEquipeStatus === 'loading' || this.qryPanoramaEquipeStatus === 'idle';

    let body;
    if (this.qryPanoramaEquipeStatus === 'error') {
      body = html`<div class="rounded-md bg-[var(--status-error-bg,transparent)] p-4" role="alert">
        <p class="text-sm font-medium text-[var(--status-error-text,currentColor)]">${m.stripErrorTitle}</p>
        ${this.qryPanoramaEquipeError?.message ? html`<p class="mt-1 text-sm text-[var(--text-default,currentColor)]">${this.qryPanoramaEquipeError.message}</p>` : nothing}
        <div class="mt-2">
          <grouptriggeraction--ml-button-standard data-variant="secondary" size="sm" @action=${() => this.qryPanoramaEquipe()}>
            <Label>${m.retry}</Label>
          </grouptriggeraction--ml-button-standard>
        </div>
      </div>`;
    } else if (loading || !t) {
      body = html`<div class="h-10 w-full animate-pulse rounded-md bg-[var(--surface-alt-bg,transparent)]" aria-busy="true"></div>`;
    } else {
      body = html`
        <dl class="flex flex-wrap items-baseline gap-x-10 gap-y-3">
          <div>
            <dt class="text-xs text-[var(--text-muted,currentColor)]">${m.stripOpen}</dt>
            <dd class="text-lg font-medium tabular-nums text-[var(--text-default,currentColor)]">${formatNumber(t.abertas)}</dd>
          </div>
          <div>
            <dt class="text-xs text-[var(--text-muted,currentColor)]">${m.stripChecklist}</dt>
            <dd class="text-lg font-medium tabular-nums text-[var(--text-default,currentColor)]">
              ${t.progressoChecklist}
              <span class="text-xs font-normal text-[var(--text-muted,currentColor)]">(${formatNumber(t.itensChecklistFeitos)} ${m.stripChecklistOf} ${formatNumber(t.itensChecklistTotal)} ${m.stripItems})</span>
            </dd>
          </div>
        </dl>
        <div class="mt-4">
          <p class="text-xs text-[var(--text-muted,currentColor)]">${m.stripLists}</p>
          ${this.listas.length === 0
            ? html`<p class="mt-1 text-sm text-[var(--text-muted,currentColor)]">${m.noLists}</p>`
            : html`<ul class="mt-1 flex flex-wrap gap-x-5 gap-y-1">
                ${this.listas.map((l) => html`<li>
                  <button type="button"
                    class="inline-flex min-h-8 items-center gap-1.5 rounded px-1 text-sm text-[var(--text-default,currentColor)] hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--focus-ring,currentColor)]"
                    @click=${() => this.abrirTarefasEquipe(l.id)}>
                    <span>${l.nome}</span>
                    <span class="tabular-nums text-[var(--text-muted,currentColor)]">${formatNumber(l.details?.details?.quantidadeTarefasAbertas ?? 0)}</span>
                  </button>
                </li>`)}
              </ul>`}
        </div>
        <div class="mt-5 flex flex-wrap gap-6 border-t border-[var(--border-subtle,currentColor)] pt-3">
          <button type="button"
            class="rounded text-sm font-medium text-[var(--link-text,currentColor)] hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--focus-ring,currentColor)]"
            @click=${() => this.abrirTarefasEquipe(null)}>${m.seeAllTasks} →</button>
          <button type="button"
            class="rounded text-sm font-medium text-[var(--link-text,currentColor)] hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[var(--focus-ring,currentColor)]"
            @click=${() => this.abrirListas(null)}>${m.seeLists} →</button>
        </div>`;
    }

    return html`<section data-organism-id="destaquesEquipe" class="rounded-lg border border-[var(--border-subtle,currentColor)] bg-[var(--surface-bg,transparent)] p-5" aria-busy=${loading ? 'true' : 'false'}>
      <h2 class="mb-3 text-sm font-semibold text-[var(--text-muted,currentColor)]">${m.stripTitle}</h2>
      ${body}
    </section>`;
  }

  render() {
    this.msg = pageMessages[this.getMessageKey(pageMessages)];
    const m = this.msg;
    const showLaneHeader = this.pageStatus !== 'empty';
    return html`
      <div class="min-h-full bg-[var(--page-bg,transparent)] text-[var(--text-default,currentColor)]">
        <div class="mx-auto max-w-5xl space-y-6 px-8 py-8">
          <header class="space-y-3">
            <h1 class="text-sm font-medium uppercase tracking-wide text-[var(--text-muted,currentColor)]">${m.pageTitle}</h1>
            ${this.renderHeadline()}
          </header>

          ${showLaneHeader
            ? html`<section data-organism-id="tarefasPedemAcao" class="space-y-3" aria-busy=${this.qryTarefasPedemAcaoStatus === 'loading' ? 'true' : 'false'}>
                <div>
                  <h2 class="text-lg font-semibold text-[var(--text-strong,currentColor)]">${m.laneTitle}</h2>
                  <p class="text-sm text-[var(--text-muted,currentColor)]">${m.laneHelp}</p>
                </div>
                <div class="rounded-lg border border-[var(--border-default,currentColor)] bg-[var(--surface-bg,transparent)] p-1">
                  ${this.renderLane()}
                </div>
              </section>`
            : html`<div data-organism-id="tarefasPedemAcao" hidden></div>`}

          ${this.renderStrip()}
        </div>
      </div>
    `;
  }
}
