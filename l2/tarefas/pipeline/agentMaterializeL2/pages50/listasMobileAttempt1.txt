type PageMessageType = typeof pageMessage_pt_br;
const pageMessages: Record<string, PageMessageType> = { 'pt-br': pageMessage_pt_br };

const formatInt = (n: number) => new Intl.NumberFormat(document.documentElement.lang || undefined).format(n);

@customElement('tarefas--web--mobile--page11--listas-102047')
export class TarefasMobilePage11ListasPage extends TarefasListasShared {
  private msg: PageMessageType = pageMessage_pt_br;

  private tipoLabel(tipo: ListaTipo | null) {
    if (tipo === 'project') return this.msg['tipoProject'];
    if (tipo === 'area') return this.msg['tipoArea'];
    return '';
  }

  private errorText(error: ErrorState) {
    if (!error) return '';
    if (error.code === 'client.preconditionMissing') return this.msg['errPrecondition'];
    if (error.code === 'client.selectionInvalid') return this.msg['errSelection'];
    return error.message || this.msg['errGeneric'];
  }

  private segmented<T extends string | boolean>(
    label: string,
    helper: string,
    options: { value: T; label: string }[],
    current: T | null,
    onPick: (v: T) => void
  ) {
    return html`
      <div role="group" aria-label=${label} class="flex flex-col gap-1.5">
        <span class="text-sm font-medium text-[var(--text-strong,currentColor)]">${label}</span>
        <div class="grid grid-cols-2 gap-2">
          ${options.map(
            (o) => html`
              <button
                type="button"
                aria-pressed=${current === o.value ? 'true' : 'false'}
                @click=${() => onPick(o.value)}
                class="min-h-[44px] rounded-lg border px-3 text-base focus-visible:ring-2 focus-visible:ring-[var(--focus-ring,currentColor)] ${current === o.value
                  ? 'bg-[var(--selected-bg,transparent)] text-[var(--selected-text,currentColor)] border-[var(--selected-border,currentColor)] font-semibold'
                  : 'bg-[var(--surface-bg,transparent)] text-[var(--text-default,currentColor)] border-[var(--border-default,currentColor)]'}"
              >
                ${o.label}
              </button>
            `
          )}
        </div>
        <span class="text-xs text-[var(--text-muted,currentColor)]">${helper}</span>
      </div>
    `;
  }

  private renderFields(
    draft: { nome: string | null; details: { tipo: ListaTipo | null; descricao: string | null }; ativa: boolean | null },
    update: (next: any) => void
  ) {
    return html`
      <groupentertext--ml-floating-text-input
        data-class="w-full"
        required
        .value=${draft.nome ?? ''}
        @input=${(e: CustomEvent<{ value: string }>) => update({ ...draft, nome: e.detail.value })}
        @change=${(e: CustomEvent<{ value: string }>) => update({ ...draft, nome: e.detail.value })}
      >
        <Label>${this.msg['fieldNome']}</Label>
        <Helper>${this.msg['nomeHelper']}</Helper>
      </groupentertext--ml-floating-text-input>

      ${this.segmented<ListaTipo>(
        this.msg['fieldTipo'],
        this.msg['tipoHelper'],
        [
          { value: 'project', label: this.msg['tipoProject'] },
          { value: 'area', label: this.msg['tipoArea'] }
        ],
        draft.details.tipo,
        (v) => update({ ...draft, details: { ...draft.details, tipo: v } })
      )}

      <groupentertext--ml-multiline-text
        data-class="w-full"
        rows="4"
        maxLength="500"
        placeholder=${this.msg['descricaoPlaceholder']}
        .value=${draft.details.descricao ?? ''}
        @input=${(e: CustomEvent<{ value: string }>) =>
          update({ ...draft, details: { ...draft.details, descricao: e.detail.value } })}
        @change=${(e: CustomEvent<{ value: string }>) =>
          update({ ...draft, details: { ...draft.details, descricao: e.detail.value } })}
      >
        <Label>${this.msg['fieldDescricao']}</Label>
      </groupentertext--ml-multiline-text>

      ${this.segmented<boolean>(
        this.msg['fieldAtiva'],
        this.msg['ativaHelper'],
        [
          { value: true, label: this.msg['ativa'] },
          { value: false, label: this.msg['inativa'] }
        ],
        draft.ativa,
        (v) => update({ ...draft, ativa: v })
      )}
    `;
  }

  private renderCreateForm() {
    const d = this.criarListaDraft;
    const incomplete = !(d.nome ?? '').trim() || d.details.tipo === null || d.ativa === null;
    return html`
      <div data-organism-id="formularioLista" class="flex flex-col gap-5 px-4 pb-8 pt-2">
        ${this.renderFields(d, (next) => this.setCriarListaDraft(next))}
        <div aria-live="polite">
          ${this.criarListaError
            ? html`<groupnotifyuser--ml-contextual-feedback type="error" .visible=${true} .dismissible=${false}>
                <Message>${this.criarListaError.message || this.msg['errCreate']}</Message>
              </groupnotifyuser--ml-contextual-feedback>`
            : nothing}
        </div>
        <grouptriggeraction--ml-button-standard
          size="lg"
          data-class="w-full"
          ?disabled=${incomplete}
          .loading=${this.criarListaStatus === 'loading'}
          @action=${() => {
            this.criarLista().then(() => {
              if (this.criarListaStatus === 'success') this.setScenario('lista');
            });
          }}
        >
          <Label>${this.msg['createAction']}</Label>
        </grouptriggeraction--ml-button-standard>
      </div>
    `;
  }

  private renderEditForm() {
    const d = this.salvarListaDraft;
    const incomplete = !(d.nome ?? '').trim() || d.details.tipo === null || d.ativa === null || d.id === null;
    return html`
      <div data-organism-id="formularioLista" class="flex flex-col gap-5 px-4 pb-8 pt-2">
        ${this.formularioLista === null
          ? nothing
          : html`
              <p class="text-sm text-[var(--text-muted,currentColor)]">
                ${this.msg['openTasks']}: ${formatInt(this.formularioLista.details.quantidadeTarefasAbertas)}
              </p>
            `}
        ${this.renderFields(d, (next) => this.setSalvarListaDraft(next))}
        <div aria-live="polite">
          ${this.salvarListaError
            ? html`<groupnotifyuser--ml-contextual-feedback type="error" .visible=${true} .dismissible=${false}>
                <Message>${this.salvarListaError.message || this.msg['errSave']}</Message>
              </groupnotifyuser--ml-contextual-feedback>`
            : nothing}
        </div>
        <grouptriggeraction--ml-button-standard
          size="lg"
          data-class="w-full"
          ?disabled=${incomplete}
          .loading=${this.salvarListaStatus === 'loading'}
          @action=${() => {
            this.salvarLista().then(() => {
              if (this.salvarListaStatus === 'success') {
                this.setScenario('lista');
                this.selectLista(null);
              }
            });
          }}
        >
          <Label>${this.msg['saveAction']}</Label>
        </grouptriggeraction--ml-button-standard>
      </div>
    `;
  }

  private renderList() {
    const loadError = this.listarListasError || this.buscarListasError;
    const loading = this.pageStatus === 'loading' || this.buscarListasStatus === 'loading';
    return html`
      <div data-organism-id="listasCadastradas" class="flex flex-col gap-4 px-4 pb-8 pt-2">
        <groupsearchcontent--ml-search-bar
          placeholder=${this.msg['searchPlaceholder']}
          .loading=${this.buscarListasStatus === 'loading'}
          @search=${(e: CustomEvent<{ query: string }>) => this.buscarListas(e.detail.query)}
          @change=${(e: CustomEvent<{ value: string | null }>) => this.buscarListas(e.detail.value ?? '')}
          @clear=${() => this.buscarListas('')}
        >
          <Label>${this.msg['searchLabel']}</Label>
        </groupsearchcontent--ml-search-bar>

        <grouptriggeraction--ml-button-standard
          size="lg"
          data-class="w-full"
          @action=${() => {
            this.selectLista(null);
            this.setScenario('nova');
          }}
        >
          <Label>${this.msg['newList']}</Label>
        </grouptriggeraction--ml-button-standard>

        <div aria-live="polite">
          ${loadError
            ? html`<div class="flex flex-col gap-3">
                <groupnotifyuser--ml-contextual-feedback type="error" .visible=${true} .dismissible=${false}>
                  <Message>${this.errorText(loadError)}</Message>
                </groupnotifyuser--ml-contextual-feedback>
                <grouptriggeraction--ml-button-standard
                  data-variant="secondary"
                  size="lg"
                  data-class="w-full"
                  @action=${() => this.listarListas()}
                >
                  <Label>${this.msg['retry']}</Label>
                </grouptriggeraction--ml-button-standard>
              </div>`
            : nothing}
        </div>

        <groupviewdata--ml-vertical-record-list
          .loading=${loading}
          @row-click=${(e: CustomEvent<{ index: number }>) => {
            const lista = this.listas[e.detail.index];
            if (!lista) return;
            this.selectLista(lista.id);
            this.setScenario('editar');
          }}
        >
          <Columns>
            <Column field="nome" header=${this.msg['colLista']} />
            <Column field="situacao" header=${this.msg['colSituacao']} width="96px" align="center" />
            <Column field="abertas" header=${this.msg['colAbertas']} width="64px" align="right" />
          </Columns>
          <Rows>
            ${this.listas.map(
              (lista) => html`
                <Row ?selected=${lista.id === this.selectedLista}>
                  <Cell>
                    <div class="flex min-h-[44px] flex-col justify-center py-1">
                      <span class="text-base font-semibold break-words ${lista.ativa
                        ? 'text-[var(--text-strong,currentColor)]'
                        : 'text-[var(--text-muted,currentColor)]'}">${lista.nome}</span>
                      <span class="text-xs text-[var(--text-muted,currentColor)]">${this.tipoLabel(lista.details.tipo)}</span>
                    </div>
                  </Cell>
                  <Cell>
                    <span class="inline-block rounded-full px-2 py-0.5 text-xs ${lista.ativa
                      ? 'bg-[var(--status-success-bg,transparent)] text-[var(--status-success-text,currentColor)]'
                      : 'bg-[var(--status-neutral-bg,transparent)] text-[var(--status-neutral-text,currentColor)]'}">
                      ${lista.ativa ? this.msg['ativa'] : this.msg['inativa']}
                    </span>
                  </Cell>
                  <Cell>
                    <span class="text-lg font-semibold ${lista.ativa
                      ? 'text-[var(--text-strong,currentColor)]'
                      : 'text-[var(--text-muted,currentColor)]'}">${formatInt(lista.details.quantidadeTarefasAbertas)}</span>
                  </Cell>
                </Row>
              `
            )}
          </Rows>
          <Loading>
            <div class="py-10 text-center text-[var(--text-muted,currentColor)]">${this.msg['loading']}</div>
          </Loading>
          <Empty>
            <div class="px-4 py-10 text-center text-[var(--text-muted,currentColor)]">
              ${this.busca ? this.msg['emptySearch'] : this.msg['emptyAll']}
            </div>
          </Empty>
        </groupviewdata--ml-vertical-record-list>

        <groupnotifyuser--ml-toast-notification
          type="success"
          position="bottom"
          .duration=${3500}
          .visible=${this.salvarListaStatus === 'success' || this.criarListaStatus === 'success'}
        >
          <Message>${this.salvarListaStatus === 'success' ? this.msg['savedToast'] : this.msg['createdToast']}</Message>
        </groupnotifyuser--ml-toast-notification>
      </div>
    `;
  }

  render() {
    this.msg = pageMessages[this.getMessageKey(pageMessages)];
    return html`
      <div class="min-h-full w-full bg-[var(--page-bg,transparent)] text-[var(--text-default,currentColor)]">
        <molecules--ml-scenary-102020
          mode="scenary"
          backLabel=${this.msg['back']}
          .value=${this.scenary || (this.selectedLista !== null ? 'editar' : 'lista')}
          @change=${(e: CustomEvent<{ value: string }>) => {
            if (e.target === e.currentTarget) {
              this.setScenario(e.detail.value);
              if (e.detail.value === 'lista') this.selectLista(null);
            }
          }}
        >
          <Scene value="lista" title=${this.msg['pageTitle']}>${this.renderList()}</Scene>
          <Scene value="editar" nav="back" backTo="lista" title=${this.formularioLista?.nome || this.msg['editTitleFallback']}>
            ${this.renderEditForm()}
          </Scene>
          <Scene value="nova" nav="back" backTo="lista" title=${this.msg['createTitle']}>
            ${this.renderCreateForm()}
          </Scene>
        </molecules--ml-scenary-102020>
      </div>
    `;
  }
}
