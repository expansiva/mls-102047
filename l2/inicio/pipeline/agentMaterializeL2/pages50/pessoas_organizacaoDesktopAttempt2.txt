function chip(label: string, tone: Tone) {
return html`<span class="inline-flex items-center rounded-full px-2.5 py-0.5 text-xs font-medium whitespace-nowrap ${TONES[tone]}">${label}</span>`;
}
function statusLabel(msg: PageMessageType, s: StatusPlataforma) {
const map: Record<StatusPlataforma, string> = {
Active: msg.statusActive,
Inactive: msg.statusInactive,
Merged: msg.statusMerged,
Blocked: msg.statusBlocked
};
return map[s];
}
function listText(items: string[]) {
try {
return new Intl.ListFormat(document.documentElement.lang || undefined, { style: 'long', type: 'conjunction' }).format(items);
} catch {
return items.join(', ');
}
}
function rowsOf(p: unknown): PessoaResumo[] {
if (Array.isArray(p)) return p as PessoaResumo[];
const rec = p as Record<string, unknown> | null;
if (!rec) return [];
for (const key of ['items', 'rows', 'data', 'pessoas']) {
const v = rec[key];
if (Array.isArray(v)) return v as PessoaResumo[];
}
return [];
}
function hasMoreOf(p: unknown): boolean {
if (!p || Array.isArray(p)) return false;
return (p as Record<string, unknown>)['hasMore'] === true;
}

: nothing}
</span>
</TableCell>
</TableRow>

private renderPanel() {
const sel = this.selectedPessoaOrganizacao;
const p = this.pessoa;
let detail;
if (!sel) {
detail = html`<p class="text-sm text-[var(--text-muted,currentColor)]">${this.msg.selectHint}</p>`;
} else if (this.obterPessoaStatus === 'error') {
detail = html`
<div class="flex flex-col gap-3" aria-live="polite">
<p class="text-sm text-[var(--text-default,currentColor)]">${this.msg.personError}</p>
<div>
<grouptriggeraction--ml-button-standard-102040 size="sm" data-variant="secondary" @action=${() => this.obterPessoa(sel)}>
<Label>${this.msg.retry}</Label>
</grouptriggeraction--ml-button-standard-102040>
</div>
</div>`;
} else if (this.obterPessoaStatus === 'loading' || !p) {
detail = html`
<div aria-busy="true" aria-live="polite" class="flex flex-col gap-3">
<span class="sr-only">${this.msg.loadingPerson}</span>
<div class="h-6 w-2/3 rounded animate-pulse bg-[var(--surface-alt-bg,transparent)]"></div>
<div class="h-20 rounded animate-pulse bg-[var(--surface-alt-bg,transparent)]"></div>
</div>`;
} else {
detail = this.renderPessoa(p);
}
return html`
<div class="rounded-lg border border-[var(--border-default,currentColor)] bg-[var(--surface-bg,transparent)] p-5 flex flex-col gap-5">
<div data-organism-id="detalhePessoa">${detail}</div>
<div data-organism-id="andamentoLiberacao">${sel && p && this.obterPessoaStatus !== 'error' ? this.renderAndamento(p) : nothing}</div>
</div>
`;
}
private renderPessoa(p: PessoaDetalhe) {
const id = p.details.identification;
const ini = p.details.inicio;
const contatos = p.contacts.map((c) => c.details.base.contacts);
return html`
<div class="flex flex-col gap-4">
<groupviewcard--ml-profile-card-102040>
<CardHeader>
<CardTitle>${id.name}</CardTitle>
<CardDescription>${chip(statusLabel(this.msg, id.status), STATUS_TONE[id.status])}</CardDescription>
</CardHeader>
<CardContent>
<dl class="grid grid-cols-[auto_1fr] gap-x-4 gap-y-2 text-sm">
<dt class="text-[var(--text-muted,currentColor)]">${this.msg.fieldCountry}</dt>
<dd class="text-[var(--text-strong,currentColor)]">${id.countryCode}</dd>
<dt class="text-[var(--text-muted,currentColor)]">${this.msg.fieldContacts}</dt>
<dd class="text-[var(--text-strong,currentColor)]">
${contatos.length > 0
? html`<ul class="flex flex-col gap-0.5">${contatos.map((c) => html`<li>${c}</li>`)}</ul>`
: html`<span class="text-[var(--text-muted,currentColor)]">${this.msg.noContacts}</span>`}
</dd>
<dt class="text-[var(--text-muted,currentColor)]">${this.msg.fieldConsent}</dt>
<dd class="text-[var(--text-strong,currentColor)]">${ini.consentimentoExigido ? this.msg.consentRequired : this.msg.consentNotRequired}</dd>

</CardContent>
</groupviewcard--ml-profile-card-102040>
${ini.acessoCompleto
? html`<groupnotifyuser--ml-contextual-feedback-102040 type="success" .visible=${true} .dismissible=${false}>
<Message>${this.msg.completeMessage}</Message>
</groupnotifyuser--ml-contextual-feedback-102040>`
