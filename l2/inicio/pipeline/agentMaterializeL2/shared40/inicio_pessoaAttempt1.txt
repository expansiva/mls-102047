  @property({ attribute: false }) pendencias: PendenciaAcesso[] = [];

  @property({ attribute: false }) empresa: EmpresaDestaque | null = null;

  @property({ attribute: false }) pageStatus: 'idle' | 'loading' | 'empty' | 'success' | 'error' = 'idle';

  /** state qryInicioPessoaStatus — status of the request qryInicioPessoa: idle, loading, success or error */
  @property({ attribute: false }) qryInicioPessoaStatus: 'idle' | 'loading' | 'success' | 'error' = 'idle';

  private publish<M extends StateMember>(member: M, value: this[M]): void {
    (this as unknown as Record<string, unknown>)[member] = value;
    setState(`ui.inicio.inicio_pessoa.${member}`, value);
  }

  private assignState(key: string, value: unknown): void {
    const member = STATE_MEMBER_BY_KEY[key];
    if (member) (this as unknown as Record<string, unknown>)[member] = value;
  }

  public connectedCallback(): void {
    super.connectedCallback();
    this.hydrateData();
    this.resetVisit();
    subscribe([...STATE_KEYS], this);
    void this.qryInicioPessoa();
  }

  public disconnectedCallback(): void {
    unsubscribe([...STATE_KEYS], this);
    super.disconnectedCallback();
  }

  public handleIcaStateChange(key: string, value: any): void {
    if (!(key in STATE_MEMBER_BY_KEY)) {
      super.handleIcaStateChange(key, value);
      return;
    }
    if (value === undefined) return;
    this.assignState(key, value);
    this.requestUpdate();
  }

  private hydrateData(): void {
    const meuAcesso = getState('ui.inicio.inicio_pessoa.meuAcesso');
    if (meuAcesso !== undefined) this.assignState('ui.inicio.inicio_pessoa.meuAcesso', meuAcesso);
    const pendencias = getState('ui.inicio.inicio_pessoa.pendencias');
    if (pendencias !== undefined) this.assignState('ui.inicio.inicio_pessoa.pendencias', pendencias);
    const empresa = getState('ui.inicio.inicio_pessoa.empresa');
    if (empresa !== undefined) this.assignState('ui.inicio.inicio_pessoa.empresa', empresa);
  }

  private resetVisit(): void {
    this.publish('pageStatus', 'idle');
    this.publish('qryInicioPessoaStatus', 'idle');
    this.publish('qryInicioPessoaError', null);
    this.publish('scenary', '');
  }

  /** setScenario — publishes the visible scene of the page (state scenary); not persisted, never in the URL */
  public setScenario(value: string): void {
    this.publish('scenary', value);
  }

  /** request qryInicioPessoa — loads at once the person's access status, the pending items waiting for the person and the person's company; replaces meuAcesso, pendencias and empresa; no arguments (the person is the one in the session); also the method to reload the page */
  public async qryInicioPessoa(): Promise<void> {
    if (this.qryInicioPessoaStatus === 'loading') return;
    this.publish('qryInicioPessoaStatus', 'loading');
    this.publish('qryInicioPessoaError', null);
    this.publish('pageStatus', 'loading');
    try {
      const input: Input<'inicio.inicio_pessoa.qryInicioPessoa'> = {};
      const response = await execBff<Output<'inicio.inicio_pessoa.qryInicioPessoa'>>(
        'inicio.inicio_pessoa.qryInicioPessoa',
        input,
        { mode: 'silent' },
      );
      if (!response.ok || !response.data) {
        this.publish('qryInicioPessoaError', response.error);
        this.publish('qryInicioPessoaStatus', 'error');
        this.publish('pageStatus', 'error');
        return;
      }
      const output = response.data;
      this.publish('meuAcesso', output.meuAcesso);
      this.publish('pendencias', output.pendencias);
      this.publish('empresa', output.empresa ?? null);
      this.publish('qryInicioPessoaStatus', 'success');
      this.publish('pageStatus', 'success');
    } catch (error) {
      const name = error instanceof Error ? error.name : 'unknown';
      this.publish('qryInicioPessoaError', { code: 'client.unexpected', message: '', details: { name } });
      this.publish('qryInicioPessoaStatus', 'error');
      this.publish('pageStatus', 'error');
    }
  }

  /** Opens meu_cadastro carrying the id of the person's record (entry param pessoaOrganizacaoId); refuses with client.preconditionMissing when the record is not loaded */
  private openMeuCadastro(): void {
    const id = this.meuAcesso ? this.meuAcesso.id : null;
    if (id === null || id === '') {
      this.publish('qryInicioPessoaError', {
        code: 'client.preconditionMissing',
        message: '',
        details: { params: ['pessoaOrganizacaoId'] },
      });
      return;
    }
    auraNavigate(`/inicio/meu_cadastro?pessoaOrganizacaoId=${encodeURIComponent(id)}`, { basePath: '/inicio' });
  }

  /** navigate (organism resumoMeuAcesso) — opens the person's own registration page (meu_cadastro) with the id of the loaded person record; takes no argument */
  public abrirMeuCadastroResumo(): void {
    this.openMeuCadastro();
  }

  /** navigate (organism pendenciasMeuAcesso) — opens meu_cadastro so the person completes the missing minimum data (pending item completarDadosMinimos); takes no argument */
  public completarDadosMinimos(): void {
    this.openMeuCadastro();
  }

  /** navigate (organism pendenciasMeuAcesso) — opens meu_cadastro to forward the access release request to the responsible (pending item pedirLiberacao); takes no argument */
  public encaminharPedidoLiberacao(): void {
    this.openMeuCadastro();
  }
}
