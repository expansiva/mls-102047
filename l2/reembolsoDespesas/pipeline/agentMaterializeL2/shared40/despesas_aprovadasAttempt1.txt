/** Form draft of the registrarPagamento command: only what the person types (id and version come from the loaded despesa). */
export type RegistrarPagamentoDraft = {
  details: {
    dataPagamento: string | null;
  };
};

export function emptyRegistrarPagamentoDraft(): RegistrarPagamentoDraft {
  return { details: { dataPagamento: null } };
}

const LIST_PAGE_SIZE = 20;

export class ReembolsoDespesasDespesas_aprovadasShared extends StateLitElement {

  @property({ attribute: false }) despesas: Output<'reembolsoDespesas.despesas_aprovadas.listarDespesasAprovadas'>['despesas'] | null = null;

  /** state despesa — the chosen expense with its full data and the version used to register the payment; source obterDespesaAprovada.despesa; organisms despesaAprovadaDetalhe, registroPagamentoForm */
  @property({ attribute: false }) despesa: DespesaAprovadaDetalhe | null = null;

  /** state selectedDespesa — id of the expense chosen in the queue (the record itself is in despesa); source entry.params.despesaId; organisms despesaAprovadaDetalhe, despesasAprovadasLista; persisted */
  @property({ attribute: false }) selectedDespesa: string | null = null;

  /** state colaboradorId — filter of the queue by collaborator id; source entry.params.colaboradorId; organism despesasAprovadasLista; persisted */
  @property({ attribute: false }) colaboradorId: string | null = null;

  /** state categoria — filter of the queue by category; source entry.params.categoria; organism despesasAprovadasLista; persisted */
  @property({ attribute: false }) categoria: string | null = null;

  /** state page — page number of the queue last requested or loaded; source entry.params.page; organism despesasAprovadasLista; persisted */
  @property({ attribute: false }) page: number | null = null;

  @property({ attribute: false }) pageStatus: 'idle' | 'loading' | 'empty' | 'success' | 'error' = 'idle';

  /** state listarDespesasAprovadasStatus — status of the queue request */
  @property({ attribute: false }) listarDespesasAprovadasStatus: 'idle' | 'loading' | 'success' | 'error' = 'idle';

  /** state listarDespesasAprovadasError — error of the queue request, null when none */
  @property({ attribute: false }) listarDespesasAprovadasError: ErrorState = null;

  /** state obterDespesaAprovadaStatus — status of the expense detail request */
  @property({ attribute: false }) obterDespesaAprovadaStatus: 'idle' | 'loading' | 'success' | 'error' = 'idle';

  /** state obterDespesaAprovadaError — error of the expense detail request, null when none */
  @property({ attribute: false }) obterDespesaAprovadaError: ErrorState = null;

  /** state registrarPagamentoStatus — status of the payment registration command */
  @property({ attribute: false }) registrarPagamentoStatus: 'idle' | 'loading' | 'success' | 'error' = 'idle';

  /** state registrarPagamentoError — error of the payment registration command, null when none */
  @property({ attribute: false }) registrarPagamentoError: ErrorState = null;

  @property({ attribute: false }) scenary = '';

  private subscribed = false;

  private obterSequence = 0;

  private publish<M extends StateMember>(member: M, value: ReembolsoDespesasDespesas_aprovadasShared[M]): void {
    (this as unknown as Record<string, unknown>)[member] = value;
    setState(`ui.reembolsoDespesas.despesas_aprovadas.${member}`, value);
    const paramName = PERSISTED_PARAM_BY_MEMBER[member];
    if (paramName) {
      try {
        const storageKey = `reembolsoDespesas.despesas_aprovadas.${paramName}`;
        if (value === null || value === undefined) {
          window.localStorage.removeItem(storageKey);
        } else {
          window.localStorage.setItem(storageKey, String(value));
        }
      } catch {
        // storage unavailable: the param simply does not persist
      }
    }
  }

  private assignState(key: string, value: unknown): void {
    const member = STATE_MEMBER_BY_KEY[key];
    if (member) (this as unknown as Record<string, unknown>)[member] = value;
  }

  public connectedCallback(): void {
    super.connectedCallback();
    this.hydrateData();
    this.resetVisit();
    if (!this.subscribed) {
      subscribe([...STATE_KEYS], this);
      this.subscribed = true;
    }
    const despesaId = this.applyEntryParams();
    void this.listarDespesasAprovadas({ page: this.page ?? 1 });
    if (despesaId !== null) {
      void this.obterDespesaAprovada(despesaId);
    }
  }

  public disconnectedCallback(): void {
    unsubscribe([...STATE_KEYS], this);
    this.subscribed = false;
    super.disconnectedCallback();
  }

  public handleIcaStateChange(key: string, value: any): void {
    if (!STATE_MEMBER_BY_KEY[key]) {
      super.handleIcaStateChange(key, value);
      return;
    }
    if (value === undefined) return;
    this.assignState(key, value);
    this.requestUpdate();
  }

  private hydrateData(): void {
    const dataMembers: StateMember[] = ['despesas', 'despesa'];
    for (const member of dataMembers) {
      const key = `ui.reembolsoDespesas.despesas_aprovadas.${member}`;
      const stored = getState(key);
      if (stored !== undefined) this.assignState(key, stored);
    }
  }

  private resetVisit(): void {
    this.publish('pageStatus', 'idle');
    this.publish('listarDespesasAprovadasStatus', 'idle');
    this.publish('listarDespesasAprovadasError', null);
    this.publish('obterDespesaAprovadaStatus', 'idle');
    this.publish('obterDespesaAprovadaError', null);
    this.publish('registrarPagamentoStatus', 'idle');
    this.publish('registrarPagamentoError', null);
    this.publish('registrarPagamentoDraft', emptyRegistrarPagamentoDraft());
    this.publish('scenary', '');
  }

  private resetCommandResults(): void {
    this.publish('registrarPagamentoStatus', 'idle');
    this.publish('registrarPagamentoError', null);
  }

  private readParam(name: string): string | null {
    const fromUrl = new URLSearchParams(window.location.search).get(name);
    if (fromUrl !== null && fromUrl !== '') return fromUrl;
    try {
      const stored = window.localStorage.getItem(`reembolsoDespesas.despesas_aprovadas.${name}`);
      if (stored !== null && stored !== '') return stored;
    } catch {
      // storage unavailable
    }
    return null;
  }

  /** Applies the entry params to their states; returns the id of the selected expense, if any. */
  private applyEntryParams(): string | null {
    this.publish('colaboradorId', this.readParam('colaboradorId'));
    this.publish('categoria', this.readParam('categoria'));
    const rawPage = this.readParam('page');
    let pageValue: number | null = null;
    if (rawPage !== null) {
      const parsed = Number(rawPage);
      if (Number.isFinite(parsed)) pageValue = parsed;
    }
    this.publish('page', pageValue);
    const despesaId = this.readParam('despesaId');
    this.publish('selectedDespesa', despesaId);
    return despesaId;
  }

  private unexpectedError(error: unknown): ErrorState {
    const name = error instanceof Error ? error.name : 'unknown';
    return { code: 'client.unexpected', message: '', details: { name } };
  }

  /** setScenario — publishes the visible scene of the page. */
  public setScenario(value: string): void {
    this.resetCommandResults();
    this.publish('scenary', value);
  }

  /** draft — replaces the payment registration form draft (payment date); clears the previous command result. */
  public setRegistrarPagamentoDraft(value: RegistrarPagamentoDraft): void {
    this.resetCommandResults();
    this.publish('registrarPagamentoDraft', value);
  }

  /**
   * request listarDespesasAprovadas — loads the queue of approved expenses awaiting payment; redraws despesas (replace, or append the next page when append is true), page, and pageStatus.
   * It is also the method to reload, filter and page the queue. Argument change (all optional): colaboradorId and categoria set the filter (null or '' clears it) and load from the first page;
   * page is the page to load (default 1); append adds the page to the loaded items and does nothing unless despesas.hasMore is true.
   */
  public async listarDespesasAprovadas(change: { colaboradorId?: string | null; categoria?: string | null; page?: number; append?: boolean } = {}): Promise<void> {
    if (this.listarDespesasAprovadasStatus === 'loading') return;
    const append = change.append === true;
    if (append && !(this.despesas && this.despesas.hasMore)) return;
    if (change.colaboradorId !== undefined) {
      this.publish('colaboradorId', change.colaboradorId === '' ? null : change.colaboradorId);
    }
    if (change.categoria !== undefined) {
      this.publish('categoria', change.categoria === '' ? null : change.categoria);
    }
    const pageToLoad = change.page ?? 1;
    const colaboradorId = this.colaboradorId;
    const categoria = this.categoria;
    const input: Input<'reembolsoDespesas.despesas_aprovadas.listarDespesasAprovadas'> = {
      ...(colaboradorId !== null && colaboradorId !== '' ? { colaboradorId } : {}),
      details: {
        ...(categoria !== null && categoria !== '' ? { categoria } : {}),
      },
      page: pageToLoad,
      pageSize: LIST_PAGE_SIZE,
    };
    this.publish('listarDespesasAprovadasStatus', 'loading');
    this.publish('listarDespesasAprovadasError', null);
    this.publish('pageStatus', 'loading');
    try {
      const response = await execBff<Output<'reembolsoDespesas.despesas_aprovadas.listarDespesasAprovadas'>>(
        'reembolsoDespesas.despesas_aprovadas.listarDespesasAprovadas',
        input,
        { mode: 'silent' },
      );
      if (!response.ok || !response.data) {
        this.publish('listarDespesasAprovadasError', response.error ?? { code: 'client.unexpected', message: '' });
        this.publish('listarDespesasAprovadasStatus', 'error');
        this.publish('pageStatus', 'error');
        return;
      }
      const loaded = response.data.despesas;
      const previous = this.despesas;
      const next = append && previous
        ? { ...loaded, items: [...previous.items, ...loaded.items] }
        : loaded;
      this.publish('despesas', next);
      this.publish('page', loaded.page);
      this.publish('listarDespesasAprovadasStatus', 'success');
      this.publish('pageStatus', next.items.length === 0 ? 'empty' : 'success');
    } catch (error) {
      this.publish('listarDespesasAprovadasError', this.unexpectedError(error));
      this.publish('listarDespesasAprovadasStatus', 'error');
      this.publish('pageStatus', 'error');
    }
  }

  /**
   * request obterDespesaAprovada — loads the full data of one approved expense (with its version) for the detail and the payment form; selects it: publishes selectedDespesa and replaces despesa.
   * Argument id: the id of the expense picked in the queue. A newer call supersedes an older one still loading.
   */
  public async obterDespesaAprovada(id: string): Promise<void> {
    if (this.obterDespesaAprovadaStatus === 'loading' && this.selectedDespesa === id) return;
    this.obterSequence += 1;
    const sequence = this.obterSequence;
    this.publish('selectedDespesa', id);
    this.publish('obterDespesaAprovadaStatus', 'loading');
    this.publish('obterDespesaAprovadaError', null);
    try {
      const response = await execBff<Output<'reembolsoDespesas.despesas_aprovadas.obterDespesaAprovada'>>(
        'reembolsoDespesas.despesas_aprovadas.obterDespesaAprovada',
        { id },
        { mode: 'silent' },
      );
      if (sequence !== this.obterSequence) return;
      if (!response.ok || !response.data) {
        this.publish('despesa', null);
        this.publish('obterDespesaAprovadaError', response.error ?? { code: 'client.unexpected', message: '' });
        this.publish('obterDespesaAprovadaStatus', 'error');
        return;
      }
      this.publish('despesa', response.data.despesa);
      this.publish('obterDespesaAprovadaStatus', 'success');
    } catch (error) {
      if (sequence !== this.obterSequence) return;
      this.publish('despesa', null);
      this.publish('obterDespesaAprovadaError', this.unexpectedError(error));
      this.publish('obterDespesaAprovadaStatus', 'error');
    }
  }

  /**
   * select — selects the expense of the queue by id (null clears the selection): publishes selectedDespesa and loads the record into despesa through obterDespesaAprovada.
   * Clearing also empties despesa and the draft. An empty id publishes client.selectionInvalid in listarDespesasAprovadasError.
   */
  public async selectDespesaAprovadaDetalhe(id: string | null): Promise<void> {
    this.resetCommandResults();
    if (id === null) {
      this.obterSequence += 1;
      this.publish('selectedDespesa', null);
      this.publish('despesa', null);
      this.publish('obterDespesaAprovadaStatus', 'idle');
      this.publish('obterDespesaAprovadaError', null);
      this.publish('registrarPagamentoDraft', emptyRegistrarPagamentoDraft());
      return;
    }
    if (id === '') {
      this.publish('listarDespesasAprovadasError', { code: 'client.selectionInvalid', message: '', details: { id } });
      return;
    }
    this.publish('registrarPagamentoDraft', emptyRegistrarPagamentoDraft());
    await this.obterDespesaAprovada(id);
  }

  /**
   * request registrarPagamento — registers the payment of the selected approved expense using the draft payment date and the loaded version (nothing to pass);
   * redraws despesa with the returned record (paid, new version) and removes the expense from despesas (it leaves the queue without reloading), then resets the draft.
   * Errors: client.preconditionMissing when no expense is loaded, client.requiredMissing when the payment date is empty.
   */
  public async registrarPagamento(): Promise<void> {
    if (this.registrarPagamentoStatus === 'loading') return;
    const current = this.despesa;
    if (!current) {
      this.publish('registrarPagamentoError', { code: 'client.preconditionMissing', message: '', details: { stateKeys: ['ui.reembolsoDespesas.despesas_aprovadas.despesa'] } });
      this.publish('registrarPagamentoStatus', 'error');
      return;
    }
    const dataPagamento = this.registrarPagamentoDraft.details.dataPagamento;
    if (dataPagamento === null || dataPagamento === '') {
      this.publish('registrarPagamentoError', { code: 'client.requiredMissing', message: '', details: { stateKeys: ['ui.reembolsoDespesas.despesas_aprovadas.registrarPagamentoDraft'] } });
      this.publish('registrarPagamentoStatus', 'error');
      return;
    }
    const input: Input<'reembolsoDespesas.despesas_aprovadas.registrarPagamento'> = {
      id: current.id,
      version: current.version,
      details: { dataPagamento },
    };
    this.publish('registrarPagamentoStatus', 'loading');
    this.publish('registrarPagamentoError', null);
    try {
      const response = await runBlockingUiAction(signal =>
        execBff<Output<'reembolsoDespesas.despesas_aprovadas.registrarPagamento'>>(
          'reembolsoDespesas.despesas_aprovadas.registrarPagamento',
          input,
          { mode: 'blocking', signal },
        ),
      );
      if (response === undefined) {
        this.publish('registrarPagamentoError', { code: 'client.unexpected', message: '', details: { name: 'aborted' } });
        this.publish('registrarPagamentoStatus', 'error');
        return;
      }
      if (!response.ok || !response.data) {
        this.publish('registrarPagamentoError', response.error ?? { code: 'client.unexpected', message: '' });
        this.publish('registrarPagamentoStatus', 'error');
        return;
      }
      const updated = response.data.despesa;
      this.publish('despesa', updated);
      const list = this.despesas;
      if (list) {
        this.publish('despesas', { ...list, items: list.items.filter(item => item.id !== updated.id) });
      }
      this.publish('registrarPagamentoDraft', emptyRegistrarPagamentoDraft());
      this.publish('registrarPagamentoStatus', 'success');
    } catch (error) {
      this.publish('registrarPagamentoError', this.unexpectedError(error));
      this.publish('registrarPagamentoStatus', 'error');
    }
  }
}
