const statusKeys: Record<DespesaLoad['status'], keyof PageMessageType> = {
  draft: 'statusDraft',
  awaitingApproval: 'statusAwaiting',
  rejected: 'statusRejected',
  approved: 'statusApproved',
  paid: 'statusPaid'
};

function statusClass(status: DespesaLoad['status']): string {
  switch (status) {
    case 'draft':
      return 'bg-[var(--status-neutral-bg,transparent)] text-[var(--status-neutral-text,currentColor)]';
    case 'awaitingApproval':
      return 'bg-[var(--status-warning-bg,transparent)] text-[var(--status-warning-text,currentColor)]';
    case 'rejected':
      return 'bg-[var(--status-error-bg,transparent)] text-[var(--status-error-text,currentColor)]';
    case 'approved':
      return 'bg-[var(--status-info-bg,transparent)] text-[var(--status-info-text,currentColor)]';
    default:
      return 'bg-[var(--status-success-bg,transparent)] text-[var(--status-success-text,currentColor)]';
  }
}

function lang(): string | undefined {
  return document.documentElement.lang || undefined;
}

function moneyToNumber(v: string | null): number | null {
  if (v === null || v === '') return null;
  const n = Number.parseFloat(v);
  return Number.isNaN(n) ? null : n;
}

function formatMoney(v: string): string {
  const n = moneyToNumber(v);
  if (n === null) return v;
  return new Intl.NumberFormat(lang() || 'pt-BR', { style: 'currency', currency: 'BRL' }).format(n);
}

function formatDate(v: string): string {
  const d = new Date(`${v}T00:00:00`);
  if (Number.isNaN(d.getTime())) return v;
  return new Intl.DateTimeFormat(lang(), { day: '2-digit', month: 'short', year: 'numeric' }).format(d);
}

function patch(d: DespesaDraft, key: keyof DespesaDraft['details'], value: string | null): DespesaDraft {
  return { details: { ...d.details, [key]: value } };
}

function draftOf(item: DespesaLoad): DespesaDraft {
  return {
    details: {
      dataDespesa: item.details.dataDespesa,
      categoria: item.details.categoria,
      valor: item.details.valor,
      descricao: item.details.descricao
    }
  };
}

@customElement('reembolso-despesas--web--mobile--page11--minhas_despesas-102047')
export class ReembolsoDespesasMobilePage11Minhas_despesasPage extends ReembolsoDespesasMinhas_despesasShared {
  private msg: PageMessageType = pageMessage_pt;

  private renderChip(status: DespesaLoad['status']) {
    return html`<span class="inline-flex items-center rounded-full px-2.5 py-1 text-xs font-medium ${statusClass(status)}">${this.msg[statusKeys[status]]}</span>`;
  }

  private renderLista() {
    const loading = this.pageStatus === 'loading' || this.pageStatus === 'idle';
    return html`
      <div data-organism-id="listaMinhasDespesas" class="flex flex-col gap-3 pb-24">
        ${this.pageStatus === 'error'
          ? html`<div class="rounded-lg border border-[var(--border-default,currentColor)] bg-[var(--surface-bg,transparent)] p-4 flex flex-col gap-3" role="alert" aria-live="polite">
              <p class="text-[var(--text-default,currentColor)]">${this.msg['loadError']}</p>
              <grouptriggeraction--ml-button-standard data-variant="secondary" size="lg" data-class="w-full" @action=${() => { this.load(); }}>
                <Label>${this.msg['retry']}</Label>
              </grouptriggeraction--ml-button-standard>
            </div>`
          : html`
            ${this.loadMinhasDespesasError
              ? html`<p class="text-[var(--text-default,currentColor)] rounded-lg border border-[var(--border-default,currentColor)] p-3" role="alert" aria-live="polite">${this.msg['commandFailed']} ${this.loadMinhasDespesasError.message}</p>`
              : nothing}
            <groupviewdata--ml-vertical-record-list
              .loading=${loading}
              .hoverable=${true}
              @row-click=${(e: CustomEvent<{ index: number }>) => {
                const item = this.minhasDespesas[e.detail.index];
                if (!item) return;
                this.selectDetalheDespesa(item.id);
                this.setScenario('detalhe');
              }}>
              <Columns>
                <Column field="dataDespesa" header=${this.msg['colDespesa']} />
                <Column field="status" header=${this.msg['colSituacao']} />
                <Column field="valor" header=${this.msg['colValor']} align="right" />
              </Columns>
              <Rows>
                ${this.minhasDespesas.map(
                  (d) => html`
                    <Row ?selected=${d.id === this.detalheDespesa}>
                      <Cell>
                        <div class="min-w-0">
                          <div class="text-sm font-medium text-[var(--text-strong,currentColor)]">${formatDate(d.details.dataDespesa)}</div>
                          <div class="text-xs text-[var(--text-muted,currentColor)] truncate">${d.details.categoria}</div>
                        </div>
                      </Cell>
                      <Cell>${this.renderChip(d.status)}</Cell>
                      <Cell>
                        <span class="block text-right font-semibold tabular-nums whitespace-nowrap text-[var(--text-strong,currentColor)]">${formatMoney(d.details.valor)}</span>
                      </Cell>
                    </Row>`
                )}
              </Rows>
              <Empty>
                <div class="text-center py-10 text-[var(--text-muted,currentColor)]">${this.msg['emptyList']}</div>
              </Empty>
              <Loading>
                <div class="flex justify-center py-10">
                  <groupshowprogress--ml-indeterminate-spinner size="md" label=${this.msg['loadingList']}></groupshowprogress--ml-indeterminate-spinner>
                </div>
              </Loading>
            </groupviewdata--ml-vertical-record-list>
            ${this.hasMoreListaMinhasDespesas
              ? html`<grouptriggeraction--ml-button-standard
                  data-variant="secondary"
                  size="lg"
                  data-class="w-full"
                  ?loading=${this.loadMinhasDespesasStatus === 'loading'}
                  @action=${() => { this.loadMoreListaMinhasDespesas(); }}>
                  <Label>${this.msg['loadMore']}</Label>
                </grouptriggeraction--ml-button-standard>`
              : nothing}`}
      </div>
      <div class="sticky bottom-0 -mx-4 px-4 py-3 bg-[var(--surface-bg,transparent)] border-t border-[var(--border-subtle,currentColor)]">
        <grouptriggeraction--ml-button-standard
          size="lg"
          data-class="w-full"
          @action=${() => { this.setScenario('registrar'); }}>
          <Label>${this.msg['registrarAction']}</Label>
        </grouptriggeraction--ml-button-standard>
      </div>`;
  }

  private renderForm(kind: 'registrar' | 'corrigir') {
    const isReg = kind === 'registrar';
    const cur = isReg ? this.registrarDespesaInput : this.corrigirDespesaInput;
    const d = cur.details;
    const status = isReg ? this.registrarDespesaStatus : this.corrigirDespesaStatus;
    const error = isReg ? this.registrarDespesaError : this.corrigirDespesaError;
    const busy = status === 'loading';
    const set = (key: keyof DespesaDraft['details'], value: string | null) => {
      const next = patch(cur, key, value);
      if (isReg) this.setRegistrarDespesaInput(next);
      else this.setCorrigirDespesaInput(next);
    };
    const submit = () => {
      if (isReg) {
        this.registrarDespesa().then(() => {
          if (this.registrarDespesaStatus === 'success') this.setScenario('detalhe');
        });
      } else {
        this.corrigirDespesa().then(() => {
          if (this.corrigirDespesaStatus === 'success') this.setScenario('detalhe');
        });
      }
    };
    return html`
      <div data-organism-id="formularioDespesa" class="flex flex-col gap-4 pb-6" aria-busy=${busy ? 'true' : 'false'}>
        <groupenterdate--ml-date-picker
          .value=${d.dataDespesa}
          locale="pt-BR"
          placeholder=${this.msg['phData']}
          required
          ?disabled=${busy}
          @change=${(e: CustomEvent<{ value: string | null }>) => set('dataDespesa', e.detail.value)}>
          <Label>${this.msg['labelData']}</Label>
        </groupenterdate--ml-date-picker>
        <groupentertext--ml-enter-text
          .value=${d.categoria ?? ''}
          placeholder=${this.msg['phCategoria']}
          required
          ?disabled=${busy}
          @input=${(e: CustomEvent<{ value: string }>) => set('categoria', e.detail.value)}
          @change=${(e: CustomEvent<{ value: string }>) => set('categoria', e.detail.value)}>
          <Label>${this.msg['labelCategoria']}</Label>
        </groupentertext--ml-enter-text>
        <groupentermoney--ml-enter-money-br
          .value=${moneyToNumber(d.valor)}
          placeholder=${this.msg['phValor']}
          required
          ?disabled=${busy}
          @input=${(e: CustomEvent<{ value: number | null }>) => set('valor', e.detail.value === null ? null : String(e.detail.value))}
          @change=${(e: CustomEvent<{ value: number | null }>) => set('valor', e.detail.value === null ? null : String(e.detail.value))}>
          <Label>${this.msg['labelValor']}</Label>
        </groupentermoney--ml-enter-money-br>
        <groupentertext--ml-multiline-text
          .value=${d.descricao ?? ''}
          .rows=${4}
          placeholder=${this.msg['phDescricao']}
          required
          ?disabled=${busy}
          @input=${(e: CustomEvent<{ value: string }>) => set('descricao', e.detail.value)}
          @change=${(e: CustomEvent<{ value: string }>) => set('descricao', e.detail.value)}>
          <Label>${this.msg['labelDescricao']}</Label>
        </groupentertext--ml-multiline-text>
        <div aria-live="polite">
          ${error
            ? html`<p class="text-[var(--text-default,currentColor)] rounded-lg border border-[var(--border-default,currentColor)] p-3">${this.msg['commandFailed']} ${error.message}</p>`
            : nothing}
        </div>
        <grouptriggeraction--ml-button-standard size="lg" data-class="w-full" ?loading=${busy} @action=${submit}>
          <Label>${isReg ? this.msg['registrarAction'] : this.msg['saveCorrection']}</Label>
        </grouptriggeraction--ml-button-standard>
      </div>`;
  }

  private renderFeedback(visible: boolean, type: 'success' | 'error', text: string, detail: string) {
    return html`<groupnotifyuser--ml-contextual-feedback type=${type} .visible=${visible} .dismissible=${false}>
      <Message>${text} ${detail}</Message>
    </groupnotifyuser--ml-contextual-feedback>`;
  }

  private renderAcoes(item: DespesaLoad) {
    const sending = this.enviarParaAprovacaoStatus === 'loading';
    const resending = this.reenviarParaAprovacaoStatus === 'loading';
    return html`
      <div data-organism-id="acoesDespesa" class="sticky bottom-0 -mx-4 px-4 py-3 flex flex-col gap-3 bg-[var(--surface-bg,transparent)] border-t border-[var(--border-subtle,currentColor)]" aria-live="polite">
        ${this.renderFeedback(this.enviarParaAprovacaoStatus === 'success', 'success', this.msg['sentOk'], '')}
        ${this.renderFeedback(this.reenviarParaAprovacaoStatus === 'success', 'success', this.msg['resentOk'], '')}
        ${this.renderFeedback(!!this.enviarParaAprovacaoError, 'error', this.msg['commandFailed'], this.enviarParaAprovacaoError?.message ?? '')}
        ${this.renderFeedback(!!this.reenviarParaAprovacaoError, 'error', this.msg['commandFailed'], this.reenviarParaAprovacaoError?.message ?? '')}
        ${item.status === 'draft'
          ? html`<grouptriggeraction--ml-button-standard
              size="lg"
              data-class="w-full"
              ?loading=${sending}
              @action=${() => {
                this.setEnviarParaAprovacaoInput(draftOf(item));
                this.enviarParaAprovacao();
              }}>
              <Label>${this.msg['sendApproval']}</Label>
            </grouptriggeraction--ml-button-standard>`
          : nothing}
        ${item.status === 'rejected'
          ? item.details.reenvioRealizado
            ? html`<p class="text-sm text-[var(--text-muted,currentColor)]">${this.msg['alreadyResent']}</p>`
            : html`
                <grouptriggeraction--ml-button-standard
                  size="lg"
                  data-class="w-full"
                  ?loading=${resending}
                  @action=${() => {
                    this.setReenviarParaAprovacaoInput(draftOf(item));
                    this.reenviarParaAprovacao();
                  }}>
                  <Label>${this.msg['resendApproval']}</Label>
                </grouptriggeraction--ml-button-standard>
                <grouptriggeraction--ml-button-standard
                  data-variant="secondary"
                  size="lg"
                  data-class="w-full"
                  @action=${() => {
                    this.setCorrigirDespesaInput(draftOf(item));
                    this.setScenario('corrigir');
                  }}>
                  <Label>${this.msg['correctAction']}</Label>
                </grouptriggeraction--ml-button-standard>`
          : nothing}
      </div>`;
  }

  private renderDetalhe() {
    const item = this.detalheDespesaItem;
    if (!item) {
      return html`
        <div data-organism-id="detalheDespesa" class="py-10 text-center text-[var(--text-muted,currentColor)]">${this.msg['noSelection']}</div>
        <div data-organism-id="acoesDespesa"></div>`;
    }
    return html`
      <div data-organism-id="detalheDespesa" class="pb-6">
        <groupviewcard--ml-vertical-card>
          <CardHeader>
            <CardTitle><span class="tabular-nums">${formatMoney(item.details.valor)}</span></CardTitle>
            <CardDescription>${item.details.categoria}</CardDescription>
          </CardHeader>
          <CardContent>
            <dl class="flex flex-col gap-3 text-sm">
              <div>
                <dt class="text-[var(--text-muted,currentColor)]">${this.msg['labelData']}</dt>
                <dd class="text-[var(--text-strong,currentColor)]">${formatDate(item.details.dataDespesa)}</dd>
              </div>
              <div>
                <dt class="text-[var(--text-muted,currentColor)]">${this.msg['labelSituacao']}</dt>
                <dd class="mt-1">${this.renderChip(item.status)}</dd>
              </div>
              <div>
                <dt class="text-[var(--text-muted,currentColor)]">${this.msg['labelDescricao']}</dt>
                <dd class="text-[var(--text-default,currentColor)] whitespace-pre-line">${item.details.descricao}</dd>
              </div>
            </dl>
          </CardContent>
        </groupviewcard--ml-vertical-card>
      </div>
      ${this.renderAcoes(item)}`;
  }

  render() {
    this.msg = pageMessages[this.getMessageKey(pageMessages)];
    return html`
      <div class="min-h-screen w-full max-w-[430px] mx-auto px-4 pb-4 bg-[var(--page-bg,transparent)] text-[var(--text-default,currentColor)]">
        <header class="pt-5 pb-3">
          <h1 class="text-xl font-semibold text-[var(--text-strong,currentColor)]">${this.msg['pageTitle']}</h1>
        </header>
        <molecules--ml-scenary-102020
          mode="scenary"
          .value=${this.scenary || 'lista'}
          .backLabel=${this.msg['back']}
          @change=${(e: CustomEvent<{ value: string }>) => {
            if (e.target === e.currentTarget) this.setScenario(e.detail.value);
          }}>
          <Scene value="lista" title=${this.msg['sceneLista']}>${this.renderLista()}</Scene>
          <Scene value="detalhe" title=${this.msg['sceneDetalhe']} nav="back" backTo="lista">${this.renderDetalhe()}</Scene>
          <Scene value="registrar" title=${this.msg['sceneRegistrar']} nav="back" backTo="lista">${this.renderForm('registrar')}</Scene>
          <Scene value="corrigir" title=${this.msg['sceneCorrigir']} nav="back" backTo="detalhe">${this.renderForm('corrigir')}</Scene>
        </molecules--ml-scenary-102020>
      </div>`;
  }
}
